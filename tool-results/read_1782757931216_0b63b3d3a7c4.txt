  1967→  // ─── Toggle task completion (PATCH /api/leads/[id]/tasks/[taskId]) ─
  1968→  async function toggleTask(
  1969→    id: string,
  1970→    taskId: string,
  1971→    completed: boolean
  1972→  ): Promise<boolean> {
  1973→    const prevTasks = selectedLead?.id === id ? selectedLead.tasks : null;
  1974→    const task = prevTasks?.find((t) => t.id === taskId);
  1975→    const nowIso = new Date().toISOString();
  1976→
  1977→    // Optimistic update
  1978→    setSelectedLead((prev) =>
  1979→      prev && prev.id === id
  1980→        ? {
  1981→            ...prev,
  1982→            tasks: prev.tasks.map((t) =>
  1983→              t.id === taskId
  1984→                ? {
  1985→                    ...t,
  1986→                    completed,
  1987→                    completedAt: completed ? nowIso : null,
  1988→                  }
  1989→                : t
  1990→            ),
  1991→          }
  1992→        : prev
  1993→    );
  1994→
  1995→    try {
  1996→      const res = await fetch(`/api/leads/${id}/tasks/${taskId}`, {
  1997→        method: "PATCH",
  1998→        headers: { "Content-Type": "application/json" },
  1999→        body: JSON.stringify({ completed }),
  2000→      });
  2001→      if (!res.ok) {
  2002→        const data = await res.json().catch(() => ({}));
  2003→        throw new Error(data.error || "Failed to update task");
  2004→      }
  2005→      // Push synthesized activity
  2006→      if (task) {
  2007→        setSelectedLead((prev) =>
  2008→          prev && prev.id === id
  2009→            ? {
  2010→                ...prev,
  2011→                activities: [
  2012→                  {
  2013→                    id: `tmp-toggle-${nowIso}`,
  2014→                    type: completed ? "task_completed" : "task_reopened",
  2015→                    description: `Task "${task.title}" ${completed ? "completed" : "reopened"} by ${user.name}`,
  2016→                    actorName: user.name,
  2017→                    createdAt: nowIso,
  2018→                  },
  2019→                  ...prev.activities,
  2020→                ],
  2021→              }
  2022→            : prev
  2023→        );
  2024→      }
  2025→      return true;
  2026→    } catch (e: any) {
  2027→      // Revert optimistic update
  2028→      if (prevTasks) {
  2029→        setSelectedLead((prev) =>
  2030→          prev && prev.id === id ? { ...prev, tasks: prevTasks } : prev
  2031→        );
  2032→      }
  2033→      toast.error(e.message || "Failed to update task");
  2034→      return false;
  2035→    }
  2036→  }
