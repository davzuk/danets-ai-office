   208→   208→function relativeTime(iso: string): string {
   209→   209→  const d = new Date(iso);
   210→   210→  const diff = Date.now() - d.getTime();
   211→   211→  const sec = Math.floor(diff / 1000);
   212→   212→  if (sec < 60) return "just now";
   213→   213→  const min = Math.floor(sec / 60);
   214→   214→  if (min < 60) return `${min}m ago`;
   215→   215→  const hr = Math.floor(min / 60);
   216→   216→  if (hr < 24) return `${hr}h ago`;
   217→   217→  const day = Math.floor(hr / 24);
   218→   218→  if (day < 7) return `${day}d ago`;
   219→   219→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   220→   220→}
   221→   221→
   222→   222→function fullDate(iso: string): string {
   223→   223→  return new Date(iso).toLocaleString(undefined, {
   224→   224→    month: "short",
   225→   225→    day: "numeric",
   226→   226→    year: "numeric",
   227→   227→    hour: "2-digit",
   228→   228→    minute: "2-digit",
   229→   229→  });
   230→   230→}

   249→   249→/** Contextual WhatsApp message body (NOT URL-encoded). */
   250→   250→function buildWhatsAppMessage(lead: LeadContact): string {
   251→   251→  const name = lead.name?.trim() || "there";
   252→   252→  const service = lead.service?.trim();
   253→   253→  const msg = lead.message?.trim();
   254→   254→
   255→   255→  const greeting = service
   256→   256→    ? `Hi ${name}, thanks for your interest in ${service}. How can we help?`
   257→   257→    : `Hi ${name}, thanks for reaching out. How can we help?`;
   258→   258→
   259→   259→  if (!msg) return greeting;
   260→   260→  const snippet = msg.slice(0, 280);
   261→   261→  const ellipsis = msg.length > 280 ? "…" : "";
   262→   262→  return `${greeting}\n\nYou mentioned: "${snippet}${ellipsis}"`;
   263→   263→}
   264→   264→
   265→   265→/** Contextual email subject line. */
   266→   266→function buildEmailSubject(lead: LeadContact): string {
   267→   267→  const service = lead.service?.trim();
   268→   268→  return service
   269→   269→    ? `Re: Your inquiry about ${service}`
   270→   270→    : "Re: Your inquiry";
   271→   271→}
   272→   272→
   273→   273→/** Contextual email body (plain text, line-break separated). */
   274→   274→function buildEmailBody(lead: LeadContact): string {
   275→   275→  const name = lead.name?.trim() || "there";
   276→   276→  const service = lead.service?.trim();
   277→   277→  const msg = lead.message?.trim();
   278→   278→
   279→   279→  const lines: string[] = [];
   280→   280→  lines.push(`Hi ${name},`);
   281→   281→  lines.push("");
   282→   282→  lines.push(
   283→   283→    service
   284→   284→      ? `Thanks for reaching out about ${service}. We'd love to help!`
   285→   285→      : `Thanks for reaching out. We'd love to help!`
   286→   286→  );
   287→   287→  if (msg) {
   288→   288→    const snippet = msg.slice(0, 400);
   289→   289→    const ellipsis = msg.length > 400 ? "…" : "";
   290→   290→    lines.push("");
   291→   291→    lines.push("You mentioned:");
   292→   292→    lines.push(`"${snippet}${ellipsis}"`);
   293→   293→  }
   294→   294→  lines.push("");
   295→   295→  lines.push(
   296→   296→    "We'll get back to you shortly. Feel free to reply with any additional details."
   297→   297→  );
   298→   298→  lines.push("");
   299→   299→  lines.push("Best regards,");
   300→   300→  lines.push("The Danets Team");
   301→   301→  return lines.join("\n");
   302→   302→}

   304→   304→/** Strip a phone string to digits-only for wa.me (intl format, no `+`). */
   305→   305→function normalizePhoneForWhatsApp(phone: string): string {
   306→   306→  return phone.replace(/\D/g, "");
   307→   307→}
   308→   308→
   309→   309→/** Build a `https://wa.me/<digits>?text=<encoded>` URL, or null if no usable phone. */
   310→   310→function buildWhatsAppUrl(lead: LeadContact): string | null {
   311→   311→  if (!lead.phone) return null;
   312→   312→  const digits = normalizePhoneForWhatsApp(lead.phone);
   313→   313→  if (!digits) return null;
   314→   314→  return `https://wa.me/${digits}?text=${encodeURIComponent(
   315→   315→    buildWhatsAppMessage(lead)
   316→   316→  )}`;
   317→   317→}
   318→   318→
   319→   319→/** Build a `mailto:` URL with prefilled subject + body, or null if no email. */
   320→   320→function buildMailtoUrl(lead: LeadContact): string | null {
   321→   321→  if (!lead.email) return null;
   322→   322→  const subject = encodeURIComponent(buildEmailSubject(lead));
   323→   323→  const body = encodeURIComponent(buildEmailBody(lead));
   324→   324→  return `mailto:${lead.email}?subject=${subject}&body=${body}`;
   325→   325→}
   326→   326→
   327→   327→/** Build a `tel:` URL (whitespace stripped), or null if no phone. */
   328→   328→function buildTelUrl(lead: LeadContact): string | null {
   329→   329→  if (!lead.phone) return null;
   330→   330→  return `tel:${lead.phone.replace(/\s+/g, "")}`;
   331→   331→}
