   201→   201→// ─── date helpers ────────────────────────────────────────────────
   202→   202→function relativeTime(iso: string): string {
   203→   203→  const d = new Date(iso);
   204→   204→  const diff = Date.now() - d.getTime();
   205→   205→  const sec = Math.floor(diff / 1000);
   206→   206→  if (sec < 60) return "just now";
   207→   207→  const min = Math.floor(sec / 60);
   208→   208→  if (min < 60) return `${min}m ago`;
   209→   209→  const hr = Math.floor(min / 60);
   210→   210→  if (hr < 24) return `${hr}h ago`;
   211→   211→  const day = Math.floor(hr / 24);
   212→   212→  if (day < 7) return `${day}d ago`;
   213→   213→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   214→   214→}
   215→   215→
   216→   216→function fullDate(iso: string): string {
   217→   217→  return new Date(iso).toLocaleString(undefined, {
   218→   218→    month: "short",
   219→   219→    day: "numeric",
   220→   220→    year: "numeric",
   221→   221→    hour: "2-digit",
   222→   222→    minute: "2-digit",
   223→   223→  });
   224→   224→}
   225→   225→
   226→   226→function initials(name: string): string {
   227→   227→  return name
   228→   228→    .split(/\s+/)
   229→   229→    .filter(Boolean)
   230→   230→    .slice(0, 2)
   231→   231→    .map((s) => s[0]?.toUpperCase())
   232→   232→    .join("");
   233→   233→}
