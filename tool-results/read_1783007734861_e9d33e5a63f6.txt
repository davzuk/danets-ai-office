    93→// ─── Date helpers ─────────────────────────────────────────────────────
    94→
    95→function pad(n: number): string {
    96→  return String(n).padStart(2, "0");
    97→}
    98→function toISODate(d: Date): string {
    99→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
   100→}
   101→function todayStr(): string {
   102→  return toISODate(new Date());
   103→}
   104→function firstOfMonthStr(): string {
   105→  const d = new Date();
   106→  return toISODate(new Date(d.getFullYear(), d.getMonth(), 1));
   107→}
   108→function daysAgoStr(days: number): string {
   109→  const d = new Date();
   110→  d.setDate(d.getDate() - days);
   111→  return toISODate(d);
   112→}
   113→function formatMonthLabel(ym: string): string {
   114→  const [y, m] = ym.split("-");
   115→  const d = new Date(Number(y), Number(m) - 1, 1);
   116→  return d.toLocaleDateString("en-US", { month: "short", year: "2-digit" });
   117→}
   118→
   119→function formatNumber(n: number): string {
   120→  return new Intl.NumberFormat("en-US").format(n);
   121→}
   122→function formatCost(n: number): string {
   123→  return n.toLocaleString("en-US", {
   124→    minimumFractionDigits: 2,
   125→    maximumFractionDigits: 2,
   126→  });
   127→}
   128→function prettify(s: string): string {
   129→  return s
   130→    .replace(/_/g, " ")
   131→    .replace(/\b\w/g, (c) => c.toUpperCase());
   132→}

   196→export function AnalyticsView({ user: _user }: { user: AnalyticsUser }) {
   197→  const [from, setFrom] = React.useState<string>(firstOfMonthStr());

   200→  const [appliedTo, setAppliedTo] = React.useState<string>(to);
   201→
   202→  const [report, setReport] = React.useState<Report | null>(null);
   203→  const [loading, setLoading] = React.useState<boolean>(true);
   204→  const [error, setError] = React.useState<string | null>(null);
   205→
   206→  const load = React.useCallback(async (f: string, t: string) => {
   207→    setLoading(true);
   208→    setError(null);
   209→    try {
   210→      const url = `/api/analytics/report?from=${encodeURIComponent(
   211→        f
   212→      )}&to=${encodeURIComponent(t)}`;
   213→      const res = await fetch(url, { cache: "no-store" });
   214→      if (!res.ok) {
   215→        if (res.status === 401) {
   216→          setError("Unauthorized. Please sign in again.");
   217→        } else {
   218→          setError(`Failed to load report (${res.status})`);
   219→        }
   220→        return;
   221→      }
   222→      const json = (await res.json()) as Report;
   223→      setReport(json);
   224→    } catch (e: unknown) {
   225→      const msg = e instanceof Error ? e.message : "Network error";
   226→      setError(msg);
   227→    } finally {
   228→      setLoading(false);
   229→    }
   230→  }, []);
   231→
   232→  React.useEffect(() => {
   233→    load(appliedFrom, appliedTo);
   234→  }, [appliedFrom, appliedTo, load]);
   235→

   237→    if (!from || !to) {
   238→      toast.error("Please select both dates");
   239→      return;
   240→    }
   241→    if (from > to) {
   242→      toast.error("From date must be before To date");
   243→      return;
   244→    }
   245→    setAppliedFrom(from);
   246→    setAppliedTo(to);
   247→    toast.success("Report updated", {
   248→      description: `${from} → ${to}`,
   249→    });
   250→  }
   251→
   252→  function applyQuickRange(range: "month" | "30" | "90") {
   253→    const t = todayStr();
   254→    let f: string;
   255→    if (range === "month") f = firstOfMonthStr();
   256→    else if (range === "30") f = daysAgoStr(30);
   257→    else f = daysAgoStr(90);
   258→    setFrom(f);
   259→    setTo(t);
   260→    setAppliedFrom(f);
   261→    setAppliedTo(t);
   262→    toast.success("Range applied", {
   263→      description: `${f} → ${t}`,
   264→    });
   265→  }
   266→

   269→      appliedFrom
   270→    )}&to=${encodeURIComponent(appliedTo)}&format=csv`;
   271→    window.open(url, "_blank");
   272→    toast.success("CSV export started", {
   273→      description: `${appliedFrom} → ${appliedTo}`,
   274→    });
   275→  }
   276→
   277→  const summary = report?.summary;
   278→  const trend = report?.monthlyTrend ?? [];
   279→  const sources = report?.leadsBySource ?? [];
   280→  const leadStatuses = report?.leadsByStatus ?? [];
   281→  const ticketStatuses = report?.ticketsByStatus ?? [];
   282→  const features = report?.aiByFeature ?? [];

   284→  return (
   285→    <div className="space-y-6">
   286→      {/* Header */}
   287→      <div className="flex flex-col sm:flex-row sm:items-end sm:justify-between gap-3">

   289→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   290→            <BarChart3 className="h-6 w-6 text-primary" />
   291→            Analytics &amp; Reporting
   292→          </h1>
   293→          <p className="text-sm text-muted-foreground mt-1">
   294→            Comprehensive business insights and performance metrics.
   295→          </p>
   296→        </div>
   297→        <Button
   298→          variant="outline"
   299→          size="sm"
   300→          onClick={() => load(appliedFrom, appliedTo)}
   301→          disabled={loading}
   302→          className="gap-2"
   303→        >
   304→          <RefreshCw
   305→            className={loading ? "h-4 w-4 animate-spin" : "h-4 w-4"}
   306→          />
   307→          Refresh
   308→        </Button>
   309→      </div>
   310→
   311→      {/* Date Filter Toolbar */}
   312→      <Card>
   313→        <CardContent className="space-y-3">

   315→            <div className="flex flex-wrap items-end gap-3">
   316→              <div className="space-y-1.5">
   317→                <label className="text-xs font-medium text-muted-foreground uppercase tracking-wide flex items-center gap-1">
   318→                  <CalendarDays className="h-3 w-3" /> From
   319→                </label>
   320→                <Input
   321→                  type="date"
   322→                  value={from}

   324→                  className="w-[170px]"
   325→                  max={to || undefined}
   326→                />
   327→              </div>
   328→              <div className="space-y-1.5">
   329→                <label className="text-xs font-medium text-muted-foreground uppercase tracking-wide flex items-center gap-1">
   330→                  <CalendarDays className="h-3 w-3" /> To
   331→                </label>

   333→                  type="date"
   334→                  value={to}

   336→                  className="w-[170px]"

   338→                />

   341→                onClick={applyFilters}

   344→              >
   345→                <RefreshCw

   355→                  disabled={loading}
   356→                >
   357→                  This Month
   358→                </Button>
   359→                <Button
   360→                  variant="outline"
   361→                  size="sm"
   362→                  onClick={() => applyQuickRange("30")}
   363→                  disabled={loading}

   365→                  Last 30 Days
   366→                </Button>

   371→                  disabled={loading}

   373→                  Last 90 Days
   374→                </Button>

   379→              variant="default"

   393→                {appliedFrom}
   394→              </strong>{" "}
   395→              →{" "}
   396→              <strong className="text-foreground tabular-nums">
   397→                {appliedTo}
   398→              </strong>
   399→            </span>
   400→            {summary ? (
   401→              <span className="tabular-nums">
   402→                {formatNumber(summary.totalChats)} chats ·{" "}
   403→                {formatNumber(summary.totalLeads)} leads ·{" "}
   404→                {formatNumber(summary.totalTickets)} tickets
   405→              </span>
   406→            ) : null}
   407→          </div>
   408→        </CardContent>
   409→      </Card>
   410→
   411→      {error ? (
   412→        <Card className="border-destructive/40">
   413→          <CardContent className="py-6 text-sm text-destructive">
   414→            {error}
   415→          </CardContent>
   416→        </Card>
   417→      ) : null}
   418→
   419→      {/* Summary Stat Cards */}
   420→      {loading && !report ? (
   421→        <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-4">
   422→          {Array.from({ length: 6 }).map((_, i) => (
   423→            <Card key={i}>
   424→              <CardContent className="space-y-2">
   425→                <Skeleton className="h-9 w-9 rounded-lg" />
   426→                <Skeleton className="h-7 w-20" />
   427→                <Skeleton className="h-3 w-16" />
   428→              </CardContent>
   429→            </Card>
   430→          ))}
   431→        </div>
   432→      ) : (
   433→        <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-4">
   434→          <SummaryCard
   435→            icon={MessagesSquare}
   436→            label="Total Chats"
   437→            value={formatNumber(summary?.totalChats ?? 0)}
   438→            tint="bg-primary/10 text-primary"
   439→          />
   440→          <SummaryCard
   441→            icon={Users}
   442→            label="Total Leads"
   443→            value={formatNumber(summary?.totalLeads ?? 0)}
   444→            tint="bg-accent/30 text-accent-foreground"
   445→          />
   446→          <SummaryCard
   447→            icon={TrendingUp}
   448→            label="Conversion Rate"
   449→            value={`${summary?.conversionRate ?? 0}%`}
   450→            tint="bg-emerald-500/15 text-emerald-600 dark:text-emerald-400"
   451→          />
   452→          <SummaryCard
   453→            icon={TicketIcon}
   454→            label="Total Tickets"
   455→            value={formatNumber(summary?.totalTickets ?? 0)}

   480→              Monthly Trend

   790→function LeadsByStatusList({ items }: { items: StatusItem[] }) {
   791→  const total = items.reduce((a, b) => a + b.count, 0);
   792→  return (
   793→    <>
   794→      {items.map((s) => {
   795→        const pct = total > 0 ? Math.round((s.count / total) * 100) : 0;
   796→        const color = statusColor(s.status);
   797→        return (
   798→          <div key={s.status} className="space-y-1.5">
   799→            <div className="flex items-center justify-between text-sm">
   800→              <span className="flex items-center gap-2">
   801→                <span
   802→                  className="h-2.5 w-2.5 rounded-full"
   803→                  style={{ backgroundColor: color }}
   804→                />
   805→                <span className="font-medium">{prettify(s.status)}</span>
   806→              </span>
   807→              <span className="font-medium tabular-nums">
   808→                {formatNumber(s.count)}{" "}
   809→                <span className="text-muted-foreground">({pct}%)</span>
   810→              </span>
   811→            </div>
   812→            <div className="h-2 rounded-full bg-muted overflow-hidden">
   813→              <div
   814→                className="h-full rounded-full transition-all duration-500"
   815→                style={{ width: `${pct}%`, backgroundColor: color }}
   816→              />
   817→            </div>
   818→          </div>
   819→        );
   820→      })}
   821→    </>
   822→  );
