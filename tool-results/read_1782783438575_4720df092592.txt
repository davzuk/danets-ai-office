   179→   179→function relativeTime(iso: string) {
   180→   180→  try {
   181→   181→    const t = new Date(iso).getTime();
   182→   182→    const diff = Math.max(0, Date.now() - t);
   183→   183→    const mins = Math.floor(diff / 60_000);
   184→   184→    if (mins < 1) return "just now";
   185→   185→    if (mins < 60) return `${mins}m ago`;
   186→   186→    const hrs = Math.floor(mins / 60);
   187→   187→    if (hrs < 24) return `${hrs}h ago`;
   188→   188→    const days = Math.floor(hrs / 24);
   189→   189→    if (days < 30) return `${days}d ago`;
   190→   190→    return new Date(iso).toLocaleDateString(undefined, {
   191→   191→      year: "numeric",
   192→   192→      month: "short",
   193→   193→      day: "numeric",
   194→   194→    });
   195→   195→  } catch {
   196→   196→    return iso;
   197→   197→  }
   198→   198→}

   200→   200→function formatDate(iso: string) {
   201→   201→  try {
   202→   202→    return new Date(iso).toLocaleString(undefined, {
   203→   203→      year: "numeric",
   204→   204→      month: "short",
   205→   205→      day: "numeric",
   206→   206→      hour: "2-digit",
   207→   207→      minute: "2-digit",
   208→   208→    });
   209→   209→  } catch {
   210→   210→    return iso;
   211→   211→  }
   212→   212→}

   214→   214→function truncate(text: string, max = 140) {
   215→   215→  const clean = text.replace(/\s+/g, " ").trim();
   216→   216→  if (clean.length <= max) return clean;
   217→   217→  return clean.slice(0, max).trimEnd() + "…";
   218→   218→}

   220→   220→function formatBytes(bytes: number | null) {
   221→   221→  if (bytes == null) return null;
   222→   222→  if (bytes < 1024) return `${bytes} B`;
   223→   223→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   224→   224→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   225→   225→}

   317→   317→        const res = await fetch(url, { cache: "no-store" });

   319→   319→        const data = await res.json();
   320→   320→        setDocuments(data.documents || []);
   321→   321→      } catch (e: any) {
   322→   322→        toast.error(e?.message || "Could not load documents");
   323→   323→      } finally {
   324→   324→        setLoading(false);
   325→   325→        fetchInFlight.current = false;
   326→   326→      }
   327→   327→    },

   839→   839→function DocumentDetailDialog({
   840→   840→  docId,
   841→   841→  onClose,
   842→   842→  onDeleted,
   843→   843→  onUpdated,
   844→   844→}: {
   845→   845→  docId: string | null;
   846→   846→  onClose: () => void;
   847→   847→  onDeleted: () => void;
   848→   848→  onUpdated: () => void;
   849→   849→}) {
   850→   850→  const [doc, setDoc] = useState<DocumentDetail | null>(null);
   851→   851→  const [loading, setLoading] = useState(false);
   852→   852→  const [activeTab, setActiveTab] = useState<string>("content");
   853→   853→  const [titleDraft, setTitleDraft] = useState("");
   854→   854→  const [editingTitle, setEditingTitle] = useState(false);
   855→   855→  const [savingTitle, setSavingTitle] = useState(false);
   856→   856→
   857→   857→  // Q&A tab state
   858→   858→  const [question, setQuestion] = useState("");
   859→   859→  const [asking, setAsking] = useState(false);
   860→   860→  const [questions, setQuestions] = useState<DocumentQuestion[]>([]);
   861→   861→  const [expandedChunks, setExpandedChunks] = useState<Record<string, boolean>>(
   862→   862→    {}
   863→   863→  );
   864→   864→
   865→   865→  // Summary tab state
   866→   866→  const [generatingSummary, setGeneratingSummary] = useState(false);
   867→   867→
   868→   868→  // Delete-from-detail state
   869→   869→  const [deleteConfirmOpen, setDeleteConfirmOpen] = useState(false);
   870→   870→  const [deleting, setDeleting] = useState(false);
   871→   871→
   872→   872→  // Reset state when a different doc is opened
   873→   873→  useEffect(() => {
   874→   874→    if (!docId) {
   875→   875→      setDoc(null);
   876→   876→      setTitleDraft("");
   877→   877→      setEditingTitle(false);
   878→   878→      setQuestions([]);
   879→   879→      setQuestion("");
   880→   880→      setActiveTab("content");
   881→   881→      setExpandedChunks({});
   882→   882→      return;
   883→   883→    }
   884→   884→    let cancelled = false;

   889→   889→          cache: "no-store",
   890→   890→        });
   891→   891→        if (!res.ok) throw new Error("Failed to load document");
   892→   892→        const data = await res.json();
   893→   893→        if (cancelled) return;
   894→   894→        const d: DocumentDetail = data.document;
   895→   895→        setDoc(d);
   896→   896→        setTitleDraft(d.title);
   897→   897→        setQuestions(d.questions || []);
   898→   898→      } catch (e: any) {
   899→   899→        toast.error(e?.message || "Could not load document");
   900→   900→        onClose();
   901→   901→      } finally {
   902→   902→        if (!cancelled) setLoading(false);
   903→   903→      }
   904→   904→    })();
   905→   905→    return () => {
   906→   906→      cancelled = true;
   907→   907→    };
   908→   908→  }, [docId]);
   909→   909→
   910→   910→  async function saveTitle() {
   911→   911→    if (!doc) return;
   912→   912→    const next = titleDraft.trim();
   913→   913→    if (next.length < 2) {
   914→   914→      toast.error("Title must be at least 2 characters");
   915→   915→      setTitleDraft(doc.title);
   916→   916→      setEditingTitle(false);
   917→   917→      return;
   918→   918→    }
   919→   919→    if (next === doc.title) {
   920→   920→      setEditingTitle(false);
   921→   921→      return;
   922→   922→    }
   923→   923→    setSavingTitle(true);
   924→   924→    try {
   925→   925→      const res = await fetch(`/api/documents/${doc.id}`, {
   926→   926→        method: "PATCH",
   927→   927→        headers: { "Content-Type": "application/json" },
   928→   928→        body: JSON.stringify({ title: next }),
   929→   929→      });
   930→   930→      const data = await res.json().catch(() => null);
   931→   931→      if (!res.ok) throw new Error(data?.error || "Update failed");
   932→   932→      setDoc((d) => (d ? { ...d, title: next } : d));
   933→   933→      toast.success("Title updated");
   934→   934→      onUpdated();
   935→   935→    } catch (e: any) {
   936→   936→      toast.error(e?.message || "Could not update title");
   937→   937→      setTitleDraft(doc.title);
   938→   938→    } finally {
   939→   939→      setSavingTitle(false);
   940→   940→      setEditingTitle(false);
   941→   941→    }

   943→   943→
   944→   944→  async function archiveToggle() {
   945→   945→    if (!doc) return;
   946→   946→    const next: DocStatus = doc.status === "archived" ? "active" : "archived";
   947→   947→    try {
   948→   948→      const res = await fetch(`/api/documents/${doc.id}`, {
   949→   949→        method: "PATCH",
   950→   950→        headers: { "Content-Type": "application/json" },
   951→   951→        body: JSON.stringify({ status: next }),
   952→   952→      });
   953→   953→      const data = await res.json().catch(() => null);
   954→   954→      if (!res.ok) throw new Error(data?.error || "Update failed");
   955→   955→      setDoc((d) => (d ? { ...d, status: next } : d));
   956→   956→      toast.success(next === "archived" ? "Document archived" : "Document restored");
   957→   957→      onUpdated();
   958→   958→    } catch (e: any) {
   959→   959→      toast.error(e?.message || "Could not update status");
   960→   960→    }
   961→   961→  }
   962→   962→
   963→   963→  async function handleAsk() {
   964→   964→    if (!doc) return;
   965→   965→    const q = question.trim();
   966→   966→    if (q.length < 2) {
   967→   967→      toast.error("Question must be at least 2 characters");
   968→   968→      return;
   969→   969→    }
   970→   970→    setAsking(true);
   971→   971→    // Optimistically show a placeholder so the user sees the question immediately.
   972→   972→    const placeholderId = `placeholder-${Date.now()}`;
   973→   973→    setQuestions((prev) => [
   974→   974→      {
   975→   975→        id: placeholderId,
   976→   976→        question: q,
   977→   977→        answer: null,
   978→   978→        createdAt: new Date().toISOString(),
   979→   979→      },
   980→   980→      ...prev,
   981→   981→    ]);
   982→   982→    setQuestion("");
   983→   983→    try {
   984→   984→      const res = await fetch(`/api/documents/${doc.id}/ask`, {
   985→   985→        method: "POST",
   986→   986→        headers: { "Content-Type": "application/json" },
   987→   987→        body: JSON.stringify({ question: q }),
   988→   988→      });
   989→   989→      const data: AskResponse = await res.json().catch(() => null);
   990→   990→      if (!res.ok || !data) {
   991→   991→        throw new Error((data as any)?.error || "Failed to get an answer");
   992→   992→      }
   993→   993→      const isFallback =
   994→   994→        data.chunksUsed === 0 || data.answer === FALLBACK_MARKER;
   995→   995→      const enriched: DocumentQuestion = {
   996→   996→        id: data.questionId || placeholderId,
   997→   997→        question: q,
   998→   998→        answer: data.answer,
   999→   999→        createdAt: new Date().toISOString(),
  1000→  1000→        chunksUsed: data.chunksUsed,
  1001→  1001→        chunkPreview: data.chunkPreview,
  1002→  1002→        tokens: data.tokens,
  1003→  1003→        fallback: isFallback,
  1004→  1004→      };
  1005→  1005→      setQuestions((prev) =>
  1006→  1006→        prev.map((item) => (item.id === placeholderId ? enriched : item))
  1007→  1007→      );
  1008→  1008→      // Auto-expand chunks for the freshest answer
  1009→  1009→      if (data.chunkPreview && data.chunkPreview.length > 0) {
