   243→   243→function formatRelativeTime(iso: string | null): string | null {
   244→   244→  if (!iso) return null;
   245→   245→  const then = new Date(iso).getTime();
   246→   246→  if (Number.isNaN(then)) return null;
   247→   247→  const diff = Date.now() - then;
   248→   248→  const sec = Math.round(diff / 1000);
   249→   249→  if (sec < 60) return "just now";
   250→   250→  const min = Math.round(sec / 60);
   251→   251→  if (min < 60) return `${min}m ago`;
   252→   252→  const hr = Math.round(min / 60);
   253→   253→  if (hr < 24) return `${hr}h ago`;
   254→   254→  const day = Math.round(hr / 24);
   255→   255→  if (day < 30) return `${day}d ago`;
   256→   256→  return new Date(iso).toLocaleDateString();
   257→   257→}

   468→   468→  async function handleSave() {
   469→   469→    if (!provider) {
   470→   470→      toast.error("Please select a provider");
   471→   471→      return;
   472→   472→    }
   473→   473→    setSaving(true);
   474→   474→    try {
   475→   475→      const config = buildConfig();
   476→   476→      const payload: Record<string, unknown> = {
   477→   477→        category,
   478→   478→        provider,
   479→   479→        name: providerLabel(category, provider),
   480→   480→        status: integration?.status === "active" ? "active" : "inactive",
   481→   481→        config,
   482→   482→        notes: notes.trim() || null,
   483→   483→      };
   484→   484→      const res = await fetch("/api/integrations", {
   485→   485→        method: "POST",
   486→   486→        headers: { "Content-Type": "application/json" },
   487→   487→        body: JSON.stringify(payload),
   488→   488→      });
   489→   489→      const data = await res.json();
   490→   490→      if (!res.ok) throw new Error(data.error || "Failed to save integration");
   491→   491→
   492→   492→      // Refetch to get the masked config back
   493→   493→      const listRes = await fetch("/api/integrations", { method: "GET" });
   494→   494→      const listData = await listRes.json();
   495→   495→      const fresh = (listData.integrations as Integration[])?.find(
   496→   496→        (i) => i.category === category
