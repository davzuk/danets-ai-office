   187→function formatDate(iso: string) {
   188→  try {
   189→    const d = new Date(iso);
   190→    return d.toLocaleDateString(undefined, {
   191→      year: "numeric",
   192→      month: "short",
   193→      day: "numeric",
   194→    });
   195→  } catch {
   196→    return iso;
   197→  }
   198→}

   200→function relativeTime(iso: string) {
   201→  try {
   202→    const d = new Date(iso).getTime();
   203→    const now = Date.now();
   204→    const diff = Math.max(0, now - d);
   205→    const mins = Math.floor(diff / 60_000);
   206→    if (mins < 1) return "just now";
   207→    if (mins < 60) return `${mins}m ago`;
   208→    const hrs = Math.floor(mins / 60);
   209→    if (hrs < 24) return `${hrs}h ago`;
   210→    const days = Math.floor(hrs / 24);
   211→    if (days < 30) return `${days}d ago`;
   212→    return formatDate(iso);
   213→  } catch {
   214→    return formatDate(iso);
   215→  }
   216→}

   218→function truncate(text: string, max = 120) {
   219→  const clean = text.replace(/\s+/g, " ").trim();
   220→  if (clean.length <= max) return clean;
   221→  return clean.slice(0, max).trimEnd() + "…";
   222→}
