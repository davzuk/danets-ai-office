   186→function formatDate(iso: string) {
   187→  try {
   188→    const d = new Date(iso);
   189→    return d.toLocaleDateString(undefined, {
   190→      year: "numeric",
   191→      month: "short",
   192→      day: "numeric",
   193→    });
   194→  } catch {
   195→    return iso;
   196→  }
   197→}
   198→
   199→function relativeTime(iso: string) {
   200→  try {
   201→    const d = new Date(iso).getTime();
   202→    const now = Date.now();
   203→    const diff = Math.max(0, now - d);
   204→    const mins = Math.floor(diff / 60_000);
   205→    if (mins < 1) return "just now";
   206→    if (mins < 60) return `${mins}m ago`;
   207→    const hrs = Math.floor(mins / 60);
   208→    if (hrs < 24) return `${hrs}h ago`;
   209→    const days = Math.floor(hrs / 24);
   210→    if (days < 30) return `${days}d ago`;
   211→    return formatDate(iso);
   212→  } catch {
   213→    return formatDate(iso);
   214→  }
   215→}

   217→function truncate(text: string, max = 120) {
   218→  const clean = text.replace(/\s+/g, " ").trim();
   219→  if (clean.length <= max) return clean;
   220→  return clean.slice(0, max).trimEnd() + "…";
   221→}

   353→  async function handleSave() {
   354→    if (!form.title.trim() || form.title.trim().length < 2) {
   355→      toast.error("Title must be at least 2 characters");
   356→      return;
   357→    }
   358→    if (!form.content.trim()) {
   359→      toast.error("Content is required");
   360→      return;
   361→    }
   362→    if (!form.category.trim()) {
   363→      toast.error("Category is required");
   364→      return;
   365→    }
   366→    setSaving(true);
   367→    try {
   368→      const payload = {
   369→        title: form.title.trim(),
   370→        content: form.content,
   371→        category: form.category.trim(),
   372→        sourceType: form.sourceType,
   373→        status: form.status,
   374→      };
   375→      const isEdit = !!editingId;
   376→      const url = isEdit
   377→        ? `/api/knowledge-base/${editingId}`
   378→        : "/api/knowledge-base";
   379→      const method = isEdit ? "PUT" : "POST";
   380→      const res = await fetch(url, {
   381→        method,
   382→        headers: { "Content-Type": "application/json" },
   383→        body: JSON.stringify(payload),
   384→      });
   385→      const data = await res.json();
   386→      if (!res.ok) {
   387→        throw new Error(data?.error || "Save failed");
   388→      }
   389→      toast.success(isEdit ? "Item updated" : "Item created", {
   390→        description: payload.title,
   391→      });
   392→      setDialogOpen(false);
   393→      setEditingId(null);
   394→      setForm(EMPTY_FORM);
   395→      await Promise.all([loadItems({ silent: true }), loadStats()]);
   396→    } catch (e: any) {
   397→      toast.error(e?.message || "Could not save item");
   398→    } finally {
   399→      setSaving(false);
   400→    }
   401→  }

   809→            <pre className="whitespace-pre-wrap break-words text-sm leading-relaxed font-sans bg-muted/40 p-4">
   810→              {previewItem?.content}
   811→            </pre>
   812→          </ScrollArea>
   813→          <DialogFooter className="flex-row sm:justify-between gap-2">
   814→            <div className="text-xs text-muted-foreground">
   815→              Updated {previewItem && relativeTime(previewItem.updatedAt)} ·{" "}

   817→            </div>
   818→            <div className="flex gap-2">
   819→              <Button
   820→                variant="outline"
   821→                onClick={() => {
   822→                  if (previewItem) openEdit(previewItem);
   823→                  setPreviewItem(null);
   824→                }}
   825→              >
   826→                Edit
   827→              </Button>
   828→              <Button onClick={() => setPreviewItem(null)}>Close</Button>
   829→            </div>
   830→          </DialogFooter>
   831→        </DialogContent>
   832→      </Dialog>
   833→    </div>
   834→  );
