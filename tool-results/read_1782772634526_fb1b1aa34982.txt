   155→

   157→

   160→  const [loadingList, setLoadingList] = useState(true);

   182→    setLoadingList(true);

   184→      const res = await fetch("/api/quotes");
   185→      const data = await res.json();
   186→      if (data.error) throw new Error(data.error);
   187→      setQuotes(Array.isArray(data.quotes) ? data.quotes : []);
   188→    } catch (e: any) {
   189→      toast.error(e?.message || "Failed to load proposals");

   191→      setLoadingList(false);

   201→  const resolvedService =
   202→    serviceChoice === "Other" ? serviceOther.trim() : serviceChoice;

   204→  function resetForm() {
   205→    setClientName("");
   206→    setCompanyName("");
   207→    setServiceChoice("");
   208→    setServiceOther("");
   209→    setRequirements("");
   210→    setBudgetRange("");
   211→    setTimeline("");
   212→    setNotes("");
   213→  }

   215→  async function handleSubmit(e: React.FormEvent) {
   216→    e.preventDefault();
   217→
   218→    if (!clientName.trim()) {
   219→      toast.error("Client name is required");
   220→      return;
   221→    }
   222→    if (!resolvedService) {
   223→      toast.error("Please choose or type the service needed");
   224→      return;
   225→    }
   226→    if (!requirements.trim()) {
   227→      toast.error("Requirements are required");
   228→      return;
   229→    }
   230→
   231→    setSubmitting(true);
   232→    try {
   233→      const res = await fetch("/api/quotes", {
   234→        method: "POST",
   235→        headers: { "Content-Type": "application/json" },
   236→        body: JSON.stringify({
   237→          clientName: clientName.trim(),
   238→          companyName: companyName.trim() || null,
   239→          service: resolvedService,
   240→          requirements: requirements.trim(),
   241→          budgetRange: budgetRange || null,
   242→          timeline: timeline || null,
   243→          notes: notes.trim() || null,
   244→        }),
   245→      });
   246→      const data = await res.json();
   247→      if (data.error) throw new Error(data.error);
   248→
   249→      toast.success("Proposal generated");
   250→
   251→      // Show the result in the Dialog
   252→      setActiveDoc({
   253→        id: data.document.id,
   254→        title: data.document.title,
   255→        content: data.document.content,
   256→        type: data.document.type,
   257→        createdAt: data.document.createdAt,
   258→      });
   259→      setActiveTokens(data.tokens ?? null);
   260→      setActiveMeta({
   261→        clientName: data.quote?.clientName,
   262→        createdAt: data.quote?.createdAt,
   263→      });
   264→      setActiveLoading(false);
   265→      setViewOpen(true);
   266→
   267→      resetForm();
   268→    } catch (e: any) {
   269→      toast.error(e?.message || "Failed to generate proposal");
   270→    } finally {
   271→      setSubmitting(false);
   272→      // Always refresh the list — a draft may have been created even on
   273→      // partial AI failure.
   274→      loadQuotes();
   275→    }
   276→  }

   278→  // ─── Open quote detail ─────────────────────────────────────────────

   280→  async function openQuote(quote: QuoteListItem) {
   281→    setViewOpen(true);
   282→    setActiveDoc(null);
   283→    setActiveTokens(null);
   284→    setActiveLoading(true);
   285→    setActiveMeta({
   286→      clientName: quote.clientName,
   287→      createdAt: quote.createdAt,
   288→    });
   289→    try {
   290→      const res = await fetch(`/api/quotes/${quote.id}`);
   291→      const data = await res.json();
   292→      if (data.error) throw new Error(data.error);
   293→      const latest = data.quote?.documents?.[0];
   294→      if (latest) {
   295→        setActiveDoc(latest);
   296→      } else {
   297→        toast.error("No generated proposal found. Try Regenerate.");
   298→        setViewOpen(false);
   299→      }
   300→    } catch (e: any) {
   301→      toast.error(e?.message || "Failed to load proposal");
   302→      setViewOpen(false);
   303→    } finally {
   304→      setActiveLoading(false);
   305→    }
   306→  }

   308→  // ─── Regenerate ─────────────────────────────────────────────────────

   310→  async function regenerate(quote: QuoteListItem, e?: React.MouseEvent) {
   311→    e?.stopPropagation();
   312→    setRegenId(quote.id);
   313→    try {
   314→      const res = await fetch("/api/quotes/generate", {
   315→        method: "POST",
   316→        headers: { "Content-Type": "application/json" },
   317→        body: JSON.stringify({ quoteRequestId: quote.id }),
   318→      });
   319→      const data = await res.json();
   320→      if (data.error) throw new Error(data.error);
   321→
   322→      toast.success("Proposal regenerated");
   323→      setActiveDoc(data.document);
   324→      setActiveTokens(data.tokens ?? null);
   325→      setActiveMeta({
   326→        clientName: quote.clientName,
   327→        createdAt: data.document.createdAt,
   328→      });
   329→      setViewOpen(true);
   330→      loadQuotes();
   331→    } catch (err: any) {
   332→      toast.error(err?.message || "Failed to regenerate proposal");
   333→    } finally {
   334→      setRegenId(null);
   335→    }
   336→  }

   338→  // ─── Delete ─────────────────────────────────────────────────────────

   340→  async function confirmDelete() {
   341→    if (!deleteTarget) return;
   342→    setDeleting(true);
   343→    try {
   344→      const res = await fetch(`/api/quotes/${deleteTarget.id}`, {
   345→        method: "DELETE",
   346→      });
   347→      const data = await res.json();
   348→      if (data.error) throw new Error(data.error);
   349→      toast.success("Quote deleted");
   350→      setQuotes((prev) => prev.filter((q) => q.id !== deleteTarget.id));
   351→    } catch (e: any) {
   352→      toast.error(e?.message || "Failed to delete quote");
   353→    } finally {
   354→      setDeleting(false);
   355→      setDeleteTarget(null);
   356→    }
   357→  }

   361→  async function copyText(text: string) {

   363→      await navigator.clipboard.writeText(text);

   370→  function downloadMarkdown(doc: GeneratedDocument) {
   371→    const blob = new Blob([doc.content], { type: "text/markdown;charset=utf-8" });
   372→    const url = URL.createObjectURL(blob);
   373→    const a = document.createElement("a");
   374→    const safeName =
   375→      (doc.title || "proposal").replace(/[^a-z0-9-_]+/gi, "-").toLowerCase();
   376→    a.href = url;
   377→    a.download = `${safeName}.md`;
   378→    document.body.appendChild(a);
   379→    a.click();
   380→    document.body.removeChild(a);
   381→    URL.revokeObjectURL(url);
   382→    toast.success("Downloaded markdown file");
   383→  }

   386→    const toastId = "pdf-download";
   387→    try {

   394→        <ProposalPDF
   395→          title={doc.title}
   396→          clientName={client}
   397→          businessName={user.businessName}
   398→          date={dateStr}
   399→          content={doc.content}
   400→        />
   401→      );
   402→      const blob = await pdf(element).toBlob();
   403→      const url = URL.createObjectURL(blob);
   404→      const a = document.createElement("a");
   405→      const safeName =
   406→        (doc.title || "proposal").replace(/[^a-z0-9-_]+/gi, "-").toLowerCase();
   407→      a.href = url;
   408→      a.download = `${safeName}.pdf`;
   409→      document.body.appendChild(a);
   410→      a.click();
   411→      document.body.removeChild(a);
   412→      URL.revokeObjectURL(url);
   413→      toast.success("PDF downloaded successfully", { id: toastId });
   414→    } catch (e: any) {
   415→      toast.error("PDF generation failed: " + (e?.message || ""), { id: toastId });
   416→    }
   417→  }
