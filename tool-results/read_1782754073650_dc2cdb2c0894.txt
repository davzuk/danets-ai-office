   275→function formatDate(iso: string) {
   276→  try {
   277→    const d = new Date(iso);
   278→    return d.toLocaleDateString(undefined, {
   279→      year: "numeric",
   280→      month: "short",
   281→      day: "numeric",
   282→    });
   283→  } catch {
   284→    return iso;
   285→  }
   286→}

   288→function relativeTime(iso: string) {
   289→  try {
   290→    const d = new Date(iso).getTime();
   291→    const now = Date.now();
   292→    const diff = Math.max(0, now - d);
   293→    const mins = Math.floor(diff / 60_000);
   294→    if (mins < 1) return "just now";
   295→    if (mins < 60) return `${mins}m ago`;
   296→    const hrs = Math.floor(mins / 60);
   297→    if (hrs < 24) return `${hrs}h ago`;
   298→    const days = Math.floor(hrs / 24);
   299→    if (days < 30) return `${days}d ago`;
   300→    return formatDate(iso);
   301→  } catch {
   302→    return formatDate(iso);
   303→  }
   304→}

   448→function hashCategoryToKey(category: string): keyof typeof CATEGORY_PALETTE {
   449→  let hash = 0;
   450→  const str = category.toLowerCase();
   451→  for (let i = 0; i < str.length; i++) {
   452→    hash = (hash << 5) - hash + str.charCodeAt(i);
   453→    hash |= 0;
   454→  }
   455→  return CATEGORY_HASH_KEYS[Math.abs(hash) % CATEGORY_HASH_KEYS.length];
   456→}
   457→
   458→function getCategoryColorKey(
   459→  category: string
   460→): keyof typeof CATEGORY_PALETTE {
   461→  const key = category.trim().toLowerCase();
   462→  return CATEGORY_NAME_MAP[key] ?? hashCategoryToKey(category);
   463→}

   465→function getCategoryColor(category: string): CategoryColorClasses {
   466→  return CATEGORY_PALETTE[getCategoryColorKey(category)];
   467→}

   607→  async function handleSave() {
   608→    if (!form.title.trim() || form.title.trim().length < 2) {
   609→      toast.error("Title must be at least 2 characters");
   610→      return;
   611→    }
   612→    if (!form.content.trim()) {
   613→      toast.error("Content is required");
   614→      return;
   615→    }
   616→    if (!form.category.trim()) {
   617→      toast.error("Category is required");
   618→      return;
   619→    }
   620→    setSaving(true);
   621→    try {
   622→      const payload = {
   623→        title: form.title.trim(),
   624→        content: form.content,
   625→        category: form.category.trim(),
   626→        sourceType: form.sourceType,
   627→        status: form.status,
   628→      };
   629→      const isEdit = !!editingId;
   630→      const url = isEdit
   631→        ? `/api/knowledge-base/${editingId}`
   632→        : "/api/knowledge-base";
   633→      const method = isEdit ? "PUT" : "POST";
   634→      const res = await fetch(url, {
   635→        method,
   636→        headers: { "Content-Type": "application/json" },
   637→        body: JSON.stringify(payload),
   638→      });
   639→      const data = await res.json();
   640→      if (!res.ok) {
   641→        throw new Error(data?.error || "Save failed");
   642→      }
   643→      toast.success(isEdit ? "Item updated" : "Item created", {
   644→        description: payload.title,
   645→      });
   646→      setDialogOpen(false);
   647→      setEditingId(null);
   648→      setForm(EMPTY_FORM);
   649→      await Promise.all([loadItems({ silent: true }), loadStats()]);
   650→    } catch (e: any) {
   651→      toast.error(e?.message || "Could not save item");
   652→    } finally {
   653→      setSaving(false);
   654→    }
   655→  }
