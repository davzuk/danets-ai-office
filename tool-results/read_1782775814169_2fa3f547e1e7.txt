   185→function statusMeta(status: string): StatusMeta {
   186→  return (
   187→    STATUS_META[status] ?? {
   188→      label: status.charAt(0).toUpperCase() + status.slice(1),
   189→      className: "bg-muted text-muted-foreground border-transparent",
   190→      dotClassName: "bg-muted-foreground",
   191→    }
   192→  );
   193→}
   194→
   195→function paymentLabel(method: string | null): string | null {
   196→  if (!method) return null;
   197→  return PAYMENT_METHOD_LABEL[method] ?? method;
   198→}
   199→
   200→function formatMoney(n: number): string {
   201→  return n.toLocaleString("en-US", {
   202→    minimumFractionDigits: 0,
   203→    maximumFractionDigits: 0,
   204→  });
   205→}
   206→
   207→function formatNumber(n: number): string {
   208→  return new Intl.NumberFormat("en-US").format(n);
   209→}
   210→
   211→function formatDate(iso: string | null): string | null {
   212→  if (!iso) return null;
   213→  const d = new Date(iso);
   214→  if (Number.isNaN(d.getTime())) return null;
   215→  return d.toLocaleDateString("en-US", {
   216→    year: "numeric",
   217→    month: "short",
   218→    day: "numeric",
   219→  });
   220→}
   221→
   222→function usageBarClass(pct: number): string {
   223→  if (pct > 90) return "[&_[data-slot=progress-indicator]]:bg-destructive";
   224→  if (pct >= 60) return "[&_[data-slot=progress-indicator]]:bg-amber-500";
   225→  return "[&_[data-slot=progress-indicator]]:bg-primary";
   226→}
   227→
   228→function usageTone(pct: number): "green" | "amber" | "red" {
   229→  if (pct > 90) return "red";
   230→  if (pct >= 60) return "amber";
   231→  return "green";
   232→}
   233→
   234→const TONE_TEXT: Record<"green" | "amber" | "red", string> = {
   235→  green: "text-primary",
   236→  amber: "text-amber-600 dark:text-amber-400",
   237→  red: "text-destructive",
   238→};

   242→export function SubscriptionView({
   243→  user: _user,
   244→}: {
   245→  user: SubscriptionUser;
   246→}) {
   247→  const [data, setData] = useState<SubscriptionData | null>(null);
   248→  const [loading, setLoading] = useState(true);
   249→  const [error, setError] = useState<string | null>(null);
   250→  const [dialogOpen, setDialogOpen] = useState(false);
   251→  const [targetPlan, setTargetPlan] = useState<PlanInfo | null>(null);
   252→
   253→  const load = useCallback(async () => {
   254→    setLoading(true);
   255→    setError(null);
   256→    try {
   257→      const res = await fetch("/api/subscription", { cache: "no-store" });
   258→      if (!res.ok) {
   259→        if (res.status === 401) {
   260→          setError("Unauthorized. Please sign in again.");
   261→        } else {
   262→          setError(`Failed to load subscription (${res.status})`);
   263→        }
   264→        return;
   265→      }
   266→      const json = (await res.json()) as SubscriptionData;
   267→      setData(json);
   268→    } catch (e: unknown) {
   269→      const msg = e instanceof Error ? e.message : "Network error";
   270→      setError(msg);
   271→    } finally {
   272→      setLoading(false);
   273→    }
   274→  }, []);
   275→
   276→  useEffect(() => {
   277→    load();
   278→  }, [load]);
   279→
   280→  const currentPlan = useMemo<PlanInfo | null>(() => {
   281→    if (!data) return null;
   282→    if (data.plan) return data.plan;
   283→    // Fallback: find the plan matching the subscription's planName
   284→    return (
   285→      data.allPlans.find((p) => p.name === data.subscription.planName) ?? null
   286→    );
   287→  }, [data]);
   288→
   289→  function openChangeDialog(plan: PlanInfo) {
   290→    setTargetPlan(plan);
   291→    setDialogOpen(true);
   292→  }
   293→
   294→  function handleChangeSuccess() {
   295→    setDialogOpen(false);
   296→    setTargetPlan(null);
   297→    load();
   298→  }
   299→
   300→  return (
   301→    <div className="space-y-6">
   302→      {/* Header */}
   303→      <div className="flex flex-col sm:flex-row sm:items-end sm:justify-between gap-3">
   304→        <div>
   305→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   306→            <CreditCard className="h-6 w-6 text-primary" />
   307→            Subscription &amp; Billing
   308→          </h1>
   309→          <p className="text-sm text-muted-foreground mt-1">
   310→            Manage your plan and track usage.
   311→          </p>
   312→        </div>
   313→        <Button
   314→          variant="outline"
   315→          size="sm"
   316→          onClick={load}
   317→          disabled={loading}
   318→          className="gap-2"
   319→        >
   320→          <RefreshCw className={loading ? "h-4 w-4 animate-spin" : "h-4 w-4"} />
   321→          Refresh
   322→        </Button>
   323→      </div>
   324→
   325→      {/* Error state */}
   326→      {error && (
   327→        <Alert variant="destructive">
   328→          <AlertCircle className="h-4 w-4" />
   329→          <AlertTitle>Could not load subscription</AlertTitle>
   330→          <AlertDescription>{error}</AlertDescription>
   331→        </Alert>
   332→      )}
   333→
   334→      {/* Loading skeletons */}
   335→      {loading && !data && (
   336→        <div className="space-y-6">
   337→          <CurrentPlanSkeleton />
   338→          <UsageSkeleton />
   339→          <PlansGridSkeleton />
   340→        </div>
   341→      )}
   342→
   343→      {data && !error && (
   344→        <>
   345→          {/* Current plan card */}
   346→          <CurrentPlanCard
   347→            subscription={data.subscription}
   348→            plan={currentPlan}
   349→          />
   350→
   351→          {/* Usage this month */}
   352→          <UsageThisMonth usage={data.usage} plan={currentPlan} />
   353→
   354→          {/* Plan comparison */}
   355→          <PlanComparisonGrid
   356→            plans={data.allPlans}
   357→            currentPlanName={data.subscription.planName}
   358→            currentPriceUsd={currentPlan?.priceUsd ?? 0}
   359→            onSelect={openChangeDialog}
   360→          />
   361→
   362→          {/* Security note */}
   363→          <SecurityNote />
   364→        </>
   365→      )}
   366→
   367→      {/* Change plan dialog */}
   368→      <ChangePlanDialog
   369→        open={dialogOpen}
   370→        onOpenChange={setDialogOpen}
   371→        targetPlan={targetPlan}
   372→        currentPlan={currentPlan}
   373→        onConfirm={handleChangeSuccess}
   374→      />
   375→    </div>
   376→  );
   377→}

   567→          <UsageBar key={b.key} spec={b} />

   570→        <p className="text-xs text-muted-foreground flex items-center gap-2">

   578→
   579→function UsageBar({ spec }: { spec: UsageBarSpec }) {
   580→  const { used, limit, label, description } = spec;
   581→  const pct =
   582→    limit > 0 ? Math.min(100, Math.round((used / limit) * 100)) : used > 0 ? 100 : 0;
   583→  const tone = usageTone(pct);
   584→  const barClass = usageBarClass(pct);
   585→
   586→  return (
   587→    <div className="space-y-1.5">
   588→      <div className="flex items-baseline justify-between gap-2">
   589→        <div className="min-w-0">
   590→          <div className="text-sm font-medium truncate">{label}</div>
   591→          <div className="text-xs text-muted-foreground truncate">
   592→            {description}
   593→          </div>
   594→        </div>
   595→        <div className="flex items-baseline gap-1.5 shrink-0 tabular-nums">
   596→          <span className={`text-sm font-semibold ${TONE_TEXT[tone]}`}>
   597→            {formatNumber(used)}
   598→          </span>
   599→          <span className="text-xs text-muted-foreground">
   600→            / {limit > 0 ? formatNumber(limit) : "∞"}
   601→          </span>
   602→          <Badge
   603→            variant="secondary"
   604→            className={`text-[10px] h-5 ${
   605→              tone === "red"
   606→                ? "bg-destructive/15 text-destructive"
   607→                : tone === "amber"
   608→                ? "bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-300"
   609→                : "bg-accent/30 text-[#4d7c0f]"
   610→            }`}
   611→          >
   612→            {pct}%
   613→          </Badge>
   614→        </div>
   615→      </div>
   616→      <Progress value={pct} className={`h-2 ${barClass}`} />
   617→    </div>
   618→  );
   619→}

   804→function ChangePlanDialog({
   805→  open,
   806→  onOpenChange,
   807→  targetPlan,
   808→  currentPlan,
   809→  onConfirm,
   810→}: {
   811→  open: boolean;
   812→  onOpenChange: (open: boolean) => void;
   813→  targetPlan: PlanInfo | null;
   814→  currentPlan: PlanInfo | null;
   815→  onConfirm: () => void;
   816→}) {
   817→  const [paymentMethod, setPaymentMethod] =
   818→    useState<string>("paystack");
   819→  const [paymentRef, setPaymentRef] = useState<string>("");
   820→  const [saving, setSaving] = useState(false);
   821→
   822→  useEffect(() => {
   823→    if (!open) {
   824→      // Reset state when dialog closes
   825→      setPaymentMethod("paystack");
   826→      setPaymentRef("");
   827→      setSaving(false);
   828→    }
   829→  }, [open]);
   830→
   831→  const fromName = currentPlan ? planLabel(currentPlan.name) : "—";
   832→  const fromPrice = currentPlan?.priceUsd ?? 0;
   833→  const toName = targetPlan ? planLabel(targetPlan.name) : "—";
   834→  const toPrice = targetPlan?.priceUsd ?? 0;
   835→  const diff = toPrice - fromPrice;
   836→  const direction =
   837→    diff > 0 ? "upgrade" : diff < 0 ? "downgrade" : "switch";
   838→
   839→  async function handleConfirm() {
   840→    if (!targetPlan) return;
   841→    setSaving(true);
   842→    try {
   843→      const res = await fetch("/api/subscription", {
   844→        method: "PUT",
   845→        headers: { "Content-Type": "application/json" },
   846→        body: JSON.stringify({
   847→          planName: targetPlan.name,
   848→          paymentMethod,
   849→          paymentRef: paymentRef.trim() || null,
   850→          status: "active",
   851→        }),
   852→      });
   853→      if (!res.ok) {
   854→        let msg = `Failed to update plan (${res.status})`;
   855→        try {
   856→          const j = await res.json();
   857→          if (j?.error) msg = j.error;
   858→          else if (j?.message) msg = j.message;
   859→        } catch {
   860→          // ignore parse error
   861→        }
   862→        toast.error(msg);
   863→        return;
   864→      }
   865→      toast.success(`Plan changed to ${planLabel(targetPlan.name)}`, {
   866→        description:
   867→          diff > 0
   868→            ? `+$${formatMoney(diff)}/mo — effective immediately.`
   869→            : diff < 0
   870→            ? `-$${formatMoney(Math.abs(diff))}/mo — effective immediately.`
   871→            : "Your new plan is now active.",
   872→      });
   873→      onConfirm();
   874→    } catch (e: unknown) {
   875→      const msg = e instanceof Error ? e.message : "Network error";
   876→      toast.error(msg);
   877→    } finally {
   878→      setSaving(false);
   879→    }
   880→  }
   881→
   882→  return (
   883→    <Dialog open={open} onOpenChange={onOpenChange}>
   884→      <DialogContent className="sm:max-w-md">
   885→        <DialogHeader>
   886→          <DialogTitle className="flex items-center gap-2">
   887→            <CreditCard className="h-5 w-5 text-primary" />
   888→            Change Plan
   889→          </DialogTitle>
   890→          <DialogDescription>
   891→            Confirm your plan change and record a payment method.
   892→          </DialogDescription>
   893→        </DialogHeader>
   894→
   895→        {targetPlan && (
   896→          <div className="space-y-4">
   897→            {/* From → To summary */}
   898→            <div className="rounded-lg border bg-muted/40 p-4 space-y-2">
   899→              <div className="flex items-center justify-between text-sm">
   900→                <span className="text-muted-foreground">From</span>
   901→                <span className="font-medium">
   902→                  {fromName} · ${formatMoney(fromPrice)}/mo
   903→                </span>
   904→              </div>
   905→              <div className="flex items-center justify-between text-sm">
   906→                <span className="text-muted-foreground">To</span>
   907→                <span className="font-semibold">
   908→                  {toName} · ${formatMoney(toPrice)}/mo
   909→                </span>
   910→              </div>
   911→              <Separator />
   912→              <div className="flex items-center justify-between text-sm">
   913→                <span className="text-muted-foreground">Price change</span>
   914→                <Badge
   915→                  className={
   916→                    diff > 0
   917→                      ? "bg-primary text-primary-foreground"
   918→                      : diff < 0
   919→                      ? "bg-accent/30 text-[#4d7c0f]"
   920→                      : "bg-muted text-muted-foreground"
   921→                  }
   922→                >
   923→                  {diff > 0
   924→                    ? `+$${formatMoney(diff)}/mo`
   925→                    : diff < 0
   926→                    ? `-$${formatMoney(Math.abs(diff))}/mo`
   927→                    : "No change"}
   928→                </Badge>
   929→              </div>
   930→            </div>
   931→
   932→            {/* Payment method */}
   933→            <div className="space-y-1.5">
   934→              <Label htmlFor="pay-method">Payment method</Label>
   935→              <Select value={paymentMethod} onValueChange={setPaymentMethod}>
   936→                <SelectTrigger id="pay-method" className="w-full">
   937→                  <SelectValue placeholder="Choose a provider" />
   938→                </SelectTrigger>
   939→                <SelectContent>
   940→                  {PAYMENT_OPTIONS.map((opt) => (
   941→                    <SelectItem key={opt.value} value={opt.value}>
   942→                      {opt.label}
   943→                    </SelectItem>
   944→                  ))}
   945→                </SelectContent>
   946→              </Select>
   947→            </div>
   948→
   949→            {/* Payment reference */}
   950→            <div className="space-y-1.5">
   951→              <Label htmlFor="pay-ref">
   952→                Payment reference{" "}
   953→                <span className="text-muted-foreground font-normal">
   954→                  (optional)
   955→                </span>
   956→              </Label>
   957→              <Input
   958→                id="pay-ref"
   959→                placeholder="e.g. txn_abc123, transfer ref, receipt no."
   960→                value={paymentRef}
   961→                onChange={(e) => setPaymentRef(e.target.value)}
   962→                maxLength={200}
   963→              />
   964→              <p className="text-xs text-muted-foreground">
   965→                Used for manual reconciliation. Leave blank if not applicable.
   966→              </p>
   967→            </div>
   968→
   969→            {/* Manual tracking note */}
   970→            <Alert className="border-amber-300/60 bg-amber-50 text-amber-900 dark:bg-amber-900/20 dark:text-amber-200 dark:border-amber-700/40">
   971→              <Info className="h-4 w-4" />
   972→              <AlertTitle>Payment gateway integration is coming soon</AlertTitle>
   973→              <AlertDescription>
   974→                For now, plans are changed manually with payment tracking. Our
   975→                team will verify your payment before activating premium features.
   976→              </AlertDescription>
   977→            </Alert>
   978→          </div>
   979→        )}
   980→
   981→        <DialogFooter className="gap-2">
   982→          <Button
   983→            variant="outline"
   984→            onClick={() => onOpenChange(false)}
   985→            disabled={saving}
   986→          >
   987→            Cancel
   988→          </Button>
   989→          <Button onClick={handleConfirm} disabled={saving || !targetPlan}>
   990→            {saving ? (
   991→              <>
   992→                <Loader2 className="h-4 w-4 animate-spin" />
   993→                Confirming…
   994→              </>
   995→            ) : (
   996→              <>
   997→                <Check className="h-4 w-4" />
   998→                Confirm {direction === "upgrade" ? "Upgrade" : direction === "downgrade" ? "Downgrade" : "Change"}
   999→              </>
  1000→            )}
  1001→          </Button>
  1002→        </DialogFooter>
  1003→      </DialogContent>
  1004→    </Dialog>
  1005→  );
  1006→}
