   179→function relativeTime(iso: string) {
   180→  try {
   181→    const t = new Date(iso).getTime();
   182→    const diff = Math.max(0, Date.now() - t);
   183→    const mins = Math.floor(diff / 60_000);
   184→    if (mins < 1) return "just now";
   185→    if (mins < 60) return `${mins}m ago`;
   186→    const hrs = Math.floor(mins / 60);
   187→    if (hrs < 24) return `${hrs}h ago`;
   188→    const days = Math.floor(hrs / 24);
   189→    if (days < 30) return `${days}d ago`;
   190→    return new Date(iso).toLocaleDateString(undefined, {
   191→      year: "numeric",
   192→      month: "short",
   193→      day: "numeric",
   194→    });
   195→  } catch {
   196→    return iso;
   197→  }
   198→}

   200→function formatDate(iso: string) {
   201→  try {
   202→    return new Date(iso).toLocaleString(undefined, {
   203→      year: "numeric",
   204→      month: "short",
   205→      day: "numeric",
   206→      hour: "2-digit",
   207→      minute: "2-digit",
   208→    });
   209→  } catch {
   210→    return iso;
   211→  }
   212→}

   214→function truncate(text: string, max = 140) {
   215→  const clean = text.replace(/\s+/g, " ").trim();
   216→  if (clean.length <= max) return clean;
   217→  return clean.slice(0, max).trimEnd() + "…";
   218→}

   220→function formatBytes(bytes: number | null) {
   221→  if (bytes == null) return null;
   222→  if (bytes < 1024) return `${bytes} B`;
   223→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   224→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   225→}

   228→  const opt = SOURCE_TYPE_OPTIONS.find((o) => o.value === sourceType);

   233→      <Icon className="size-3" />
   234→      {label}
   235→    </Badge>
   236→  );
   237→}

   275→export function DocumentsView({ user: _user }: { user: DocUser }) {
   276→  const [documents, setDocuments] = useState<DocumentListItem[]>([]);
   277→  const [loading, setLoading] = useState(true);
   278→  const [refreshing, setRefreshing] = useState(false);
   279→
   280→  // Toolbar
   281→  const [searchInput, setSearchInput] = useState("");
   282→  const [debouncedSearch, setDebouncedSearch] = useState("");
   283→  const [categoryFilter, setCategoryFilter] = useState<string>("all");
   284→
   285→  // New document dialog
   286→  const [createOpen, setCreateOpen] = useState(false);
   287→  const [form, setForm] = useState(EMPTY_FORM);
   288→  const [creating, setCreating] = useState(false);
   289→
   290→  // Detail dialog
   291→  const [detailId, setDetailId] = useState<string | null>(null);
   292→
   293→  // Delete state
   294→  const [deleteTarget, setDeleteTarget] = useState<DocumentListItem | null>(
   295→    null
   296→  );
   297→  const [deleting, setDeleting] = useState(false);
   298→
   299→  const fetchInFlight = useRef(false);
   300→
   301→  // Debounced search (300ms)
   302→  useEffect(() => {
   303→    const t = setTimeout(() => setDebouncedSearch(searchInput.trim()), 300);
   304→    return () => clearTimeout(t);
   305→  }, [searchInput]);
   306→
   307→  const loadDocuments = useCallback(
   308→    async (opts?: { silent?: boolean }) => {
   309→      if (fetchInFlight.current) return;
   310→      fetchInFlight.current = true;
   311→      if (!opts?.silent) setLoading(true);
   312→      try {
   313→        const params = new URLSearchParams();
   314→        if (debouncedSearch) params.set("q", debouncedSearch);
   315→        if (categoryFilter !== "all") params.set("category", categoryFilter);
   316→        const url = `/api/documents${params.toString() ? `?${params}` : ""}`;
   317→        const res = await fetch(url, { cache: "no-store" });
   318→        if (!res.ok) throw new Error("Failed to load documents");
   319→        const data = await res.json();
   320→        setDocuments(data.documents || []);
   321→      } catch (e: any) {
   322→        toast.error(e?.message || "Could not load documents");
   323→      } finally {
   324→        setLoading(false);
   325→        fetchInFlight.current = false;
   326→      }
   327→    },
   328→    [debouncedSearch, categoryFilter]
   329→  );
   330→
   331→  useEffect(() => {
   332→    loadDocuments();
   333→  }, [loadDocuments]);
   334→
   335→  // Derived: unique categories from the loaded list
   336→  const categoryOptions = useMemo(() => {
   337→    const set = new Set<string>();
   338→    documents.forEach((d) => d.category && set.add(d.category));
   339→    return Array.from(set).sort((a, b) => a.localeCompare(b));
   340→  }, [documents]);
   341→
   342→  async function handleCreate() {
   343→    if (!form.title.trim() || form.title.trim().length < 2) {
   344→      toast.error("Title must be at least 2 characters");
   345→      return;
   346→    }
   347→    if (!form.category.trim()) {
   348→      toast.error("Category is required");
   349→      return;
   350→    }
   351→    if (!form.content.trim()) {
   352→      toast.error("Content is required");
   353→      return;
   354→    }
   355→    setCreating(true);
   356→    try {
   357→      const res = await fetch("/api/documents", {
   358→        method: "POST",
   359→        headers: { "Content-Type": "application/json" },
   360→        body: JSON.stringify({
   361→          title: form.title.trim(),
   362→          content: form.content,
   363→          category: form.category.trim(),
   364→          sourceType: form.sourceType,
   365→        }),
   366→      });
   367→      const data = await res.json().catch(() => null);
   368→      if (!res.ok) {
   369→        throw new Error(data?.error || "Failed to create document");
   370→      }
   371→      const newId: string | undefined = data?.document?.id;
   372→      toast.success("Document created", {
   373→        description: form.title.trim(),
   374→      });
   375→      setCreateOpen(false);
   376→      setForm(EMPTY_FORM);
   377→      await loadDocuments({ silent: true });
   378→      // Open the freshly-created document
   379→      if (newId) {
   380→        setDetailId(newId);
   381→      }
   382→    } catch (e: any) {
   383→      toast.error(e?.message || "Could not create document");
   384→    } finally {
   385→      setCreating(false);
   386→    }
   387→  }
   388→
   389→  async function handleConfirmDelete() {
   390→    if (!deleteTarget) return;
   391→    setDeleting(true);
   392→    try {
   393→      const res = await fetch(`/api/documents/${deleteTarget.id}`, {
   394→        method: "DELETE",
   395→      });
   396→      const data = await res.json().catch(() => null);
   397→      if (!res.ok) throw new Error(data?.error || "Delete failed");
   398→      toast.success("Document deleted", {
   399→        description: deleteTarget.title,
   400→      });
   401→      setDeleteTarget(null);
   402→      await loadDocuments({ silent: true });
   403→    } catch (e: any) {
   404→      toast.error(e?.message || "Could not delete document");
   405→    } finally {
   406→      setDeleting(false);
   407→    }
   408→  }
   409→

   412→    await loadDocuments({ silent: true });
   413→    setRefreshing(false);
   414→  }
   415→
   416→  const visibleCount = documents.length;
   417→  const hasFilters =
   418→    !!debouncedSearch || categoryFilter !== "all";
   419→
   420→  return (
   421→    <div className="space-y-6">
   422→      {/* Header */}
   423→      <div className="flex flex-col sm:flex-row sm:items-end sm:justify-between gap-4">

   427→              <FileText className="size-5" />
   428→            </div>
   429→            <div>
   430→              <h1 className="text-2xl font-bold tracking-tight">
   431→                AI Document Assistant
   432→              </h1>
   433→              <p className="text-sm text-muted-foreground">
   434→                Upload documents and ask AI questions based on their content.
   435→              </p>
   436→            </div>
   437→          </div>
   438→        </div>
   439→        <div className="flex items-center gap-2">
   440→          <Button
   441→            variant="outline"
   442→            size="icon"
   443→            onClick={handleRefresh}
   444→            disabled={refreshing}
   445→            title="Refresh"
   446→          >
   447→            <RefreshCw
   448→              className={cn("size-4", refreshing && "animate-spin")}
   449→            />
   450→          </Button>
   451→          <Button
   452→            onClick={() => {
   453→              setForm(EMPTY_FORM);
   454→              setCreateOpen(true);
   455→            }}
   456→            className="gap-2"
   457→          >
   458→            <Plus className="size-4" />
   459→            New Document
   460→          </Button>
   461→        </div>
   462→      </div>
   463→
   464→      {/* Toolbar */}
   465→      <Card className="py-4">
   466→        <CardContent className="px-4 py-0 space-y-3">
   467→          <div className="flex flex-col md:flex-row md:items-center gap-3">
   468→            <div className="relative flex-1 min-w-0">
   469→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   470→              <Input
   471→                value={searchInput}
   472→                onChange={(e) => setSearchInput(e.target.value)}
   473→                placeholder="Search by title or content…"
   474→                className="pl-9 pr-9"
   475→                aria-label="Search documents"
   476→              />

   478→                <button

   486→            </div>

   565→      {/* Create dialog */}

   572→      >
   573→        <DialogContent className="sm:max-w-2xl">

   576→              <FileUp className="size-5 text-primary" />
   577→              New Document
   578→            </DialogTitle>
   579→            <DialogDescription>
   580→              Paste document text and the AI will automatically split it into
   581→              searchable sections for Q&amp;A.
   582→            </DialogDescription>

   584→
   585→          <div className="grid gap-4">

   839→function DocumentDetailDialog({
   840→  docId,
   841→  onClose,
   842→  onDeleted,
   843→  onUpdated,
   844→}: {
   845→  docId: string | null;
   846→  onClose: () => void;
   847→  onDeleted: () => void;
   848→  onUpdated: () => void;
   849→}) {
   850→  const [doc, setDoc] = useState<DocumentDetail | null>(null);
   851→  const [loading, setLoading] = useState(false);
   852→  const [activeTab, setActiveTab] = useState<string>("content");
   853→  const [titleDraft, setTitleDraft] = useState("");
   854→  const [editingTitle, setEditingTitle] = useState(false);
   855→  const [savingTitle, setSavingTitle] = useState(false);
   856→
   857→  // Q&A tab state
   858→  const [question, setQuestion] = useState("");
   859→  const [asking, setAsking] = useState(false);
   860→  const [questions, setQuestions] = useState<DocumentQuestion[]>([]);
   861→  const [expandedChunks, setExpandedChunks] = useState<Record<string, boolean>>(
   862→    {}
   863→  );
   864→
   865→  // Summary tab state
   866→  const [generatingSummary, setGeneratingSummary] = useState(false);
   867→
   868→  // Delete-from-detail state
   869→  const [deleteConfirmOpen, setDeleteConfirmOpen] = useState(false);
   870→  const [deleting, setDeleting] = useState(false);
   871→
   872→  // Reset state when a different doc is opened
   873→  useEffect(() => {
   874→    if (!docId) {
   875→      setDoc(null);
   876→      setTitleDraft("");
   877→      setEditingTitle(false);
   878→      setQuestions([]);
   879→      setQuestion("");
   880→      setActiveTab("content");
   881→      setExpandedChunks({});
   882→      return;
   883→    }
   884→    let cancelled = false;
   885→    setLoading(true);
   886→    (async () => {
   887→      try {
   888→        const res = await fetch(`/api/documents/${docId}`, {
   889→          cache: "no-store",
   890→        });
   891→        if (!res.ok) throw new Error("Failed to load document");
   892→        const data = await res.json();
   893→        if (cancelled) return;
   894→        const d: DocumentDetail = data.document;
   895→        setDoc(d);
   896→        setTitleDraft(d.title);
   897→        setQuestions(d.questions || []);
   898→      } catch (e: any) {
   899→        toast.error(e?.message || "Could not load document");
   900→        onClose();
   901→      } finally {
   902→        if (!cancelled) setLoading(false);
   903→      }
   904→    })();
   905→    return () => {
   906→      cancelled = true;
   907→    };
   908→  }, [docId]);
   909→
   910→  async function saveTitle() {
   911→    if (!doc) return;
   912→    const next = titleDraft.trim();
   913→    if (next.length < 2) {
   914→      toast.error("Title must be at least 2 characters");
   915→      setTitleDraft(doc.title);
   916→      setEditingTitle(false);
   917→      return;
   918→    }
   919→    if (next === doc.title) {
   920→      setEditingTitle(false);
   921→      return;
   922→    }
   923→    setSavingTitle(true);
   924→    try {
   925→      const res = await fetch(`/api/documents/${doc.id}`, {
   926→        method: "PATCH",
   927→        headers: { "Content-Type": "application/json" },
   928→        body: JSON.stringify({ title: next }),
   929→      });
   930→      const data = await res.json().catch(() => null);
   931→      if (!res.ok) throw new Error(data?.error || "Update failed");
   932→      setDoc((d) => (d ? { ...d, title: next } : d));
   933→      toast.success("Title updated");
   934→      onUpdated();
   935→    } catch (e: any) {
   936→      toast.error(e?.message || "Could not update title");
   937→      setTitleDraft(doc.title);
   938→    } finally {
   939→      setSavingTitle(false);
   940→      setEditingTitle(false);
   941→    }
   942→  }
   943→
   944→  async function archiveToggle() {
   945→    if (!doc) return;
   946→    const next: DocStatus = doc.status === "archived" ? "active" : "archived";
   947→    try {
   948→      const res = await fetch(`/api/documents/${doc.id}`, {
   949→        method: "PATCH",
   950→        headers: { "Content-Type": "application/json" },
   951→        body: JSON.stringify({ status: next }),
   952→      });
   953→      const data = await res.json().catch(() => null);
   954→      if (!res.ok) throw new Error(data?.error || "Update failed");
   955→      setDoc((d) => (d ? { ...d, status: next } : d));
   956→      toast.success(next === "archived" ? "Document archived" : "Document restored");
   957→      onUpdated();
   958→    } catch (e: any) {
   959→      toast.error(e?.message || "Could not update status");
   960→    }
   961→  }
   962→
   963→  async function handleAsk() {
   964→    if (!doc) return;
   965→    const q = question.trim();
   966→    if (q.length < 2) {
   967→      toast.error("Question must be at least 2 characters");
   968→      return;
   969→    }
   970→    setAsking(true);
   971→    // Optimistically show a placeholder so the user sees the question immediately.
   972→    const placeholderId = `placeholder-${Date.now()}`;
   973→    setQuestions((prev) => [
   974→      {
   975→        id: placeholderId,
   976→        question: q,
   977→        answer: null,
   978→        createdAt: new Date().toISOString(),
   979→      },
   980→      ...prev,
   981→    ]);
   982→    setQuestion("");
   983→    try {
   984→      const res = await fetch(`/api/documents/${doc.id}/ask`, {
   985→        method: "POST",
   986→        headers: { "Content-Type": "application/json" },
   987→        body: JSON.stringify({ question: q }),
   988→      });
   989→      const data: AskResponse = await res.json().catch(() => null);
   990→      if (!res.ok || !data) {
   991→        throw new Error((data as any)?.error || "Failed to get an answer");
   992→      }
   993→      const isFallback =
   994→        data.chunksUsed === 0 || data.answer === FALLBACK_MARKER;
   995→      const enriched: DocumentQuestion = {
   996→        id: data.questionId || placeholderId,
   997→        question: q,
   998→        answer: data.answer,
   999→        createdAt: new Date().toISOString(),
  1000→        chunksUsed: data.chunksUsed,
  1001→        chunkPreview: data.chunkPreview,
  1002→        tokens: data.tokens,
  1003→        fallback: isFallback,
  1004→      };
  1005→      setQuestions((prev) =>
  1006→        prev.map((item) => (item.id === placeholderId ? enriched : item))
  1007→      );
  1008→      // Auto-expand chunks for the freshest answer
  1009→      if (data.chunkPreview && data.chunkPreview.length > 0) {
  1010→        setExpandedChunks((m) => ({
  1011→          ...m,
  1012→          [enriched.id]: true,
  1013→        }));
  1014→      }
  1015→      onUpdated();
  1016→    } catch (e: any) {
  1017→      // Replace placeholder with an error marker
  1018→      setQuestions((prev) =>
  1019→        prev.map((item) =>
  1020→          item.id === placeholderId
  1021→            ? {
  1022→                ...item,
  1023→                answer:
  1024→                  "Something went wrong while contacting the AI. Please try again.",
  1025→                fallback: true,
  1026→              }
  1027→            : item
  1028→        )
  1029→      );
  1030→      toast.error(e?.message || "Could not get an answer");
  1031→    } finally {
  1032→      setAsking(false);
  1033→    }
  1034→  }
  1035→
  1036→  async function handleGenerateSummary() {
  1037→    if (!doc) return;
  1038→    setGeneratingSummary(true);
  1039→    try {
  1040→      const res = await fetch(`/api/documents/${doc.id}/summary`, {
  1041→        method: "POST",
  1042→      });
  1043→      const data: SummaryResponse = await res.json().catch(() => null);
  1044→      if (!res.ok || !data) {
  1045→        throw new Error((data as any)?.error || "Failed to generate summary");
  1046→      }
  1047→      setDoc((d) =>
  1048→        d
  1049→          ? {
  1050→              ...d,
  1051→              summary: data.summary,
  1052→            }
  1053→          : d
  1054→      );
  1055→      toast.success("Summary generated", {
  1056→        description: `${data.tokens} tokens used`,
  1057→      });
  1058→      onUpdated();
  1059→    } catch (e: any) {
  1060→      toast.error(e?.message || "Could not generate summary");
  1061→    } finally {
  1062→      setGeneratingSummary(false);
  1063→    }
  1064→  }
  1065→
  1066→  async function handleDeleteFromDetail() {
  1067→    if (!doc) return;

  1079→      toast.error(e?.message || "Could not delete document");

  1081→      setDeleting(false);
  1082→    }
  1083→  }
