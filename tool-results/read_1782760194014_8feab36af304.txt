   286→// ─── date helpers ────────────────────────────────────────────────
   287→function relativeTime(iso: string | null): string {
   288→  if (!iso) return "—";
   289→  const d = new Date(iso);
   290→  const diff = Date.now() - d.getTime();
   291→  const sec = Math.floor(diff / 1000);
   292→  if (sec < 60) return "just now";
   293→  const min = Math.floor(sec / 60);
   294→  if (min < 60) return `${min}m ago`;
   295→  const hr = Math.floor(min / 60);
   296→  if (hr < 24) return `${hr}h ago`;
   297→  const day = Math.floor(hr / 24);
   298→  if (day < 7) return `${day}d ago`;
   299→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   300→}
   301→
   302→function fullDate(iso: string | null): string {
   303→  if (!iso) return "—";
   304→  return new Date(iso).toLocaleString(undefined, {
   305→    month: "short",
   306→    day: "numeric",
   307→    year: "numeric",
   308→    hour: "2-digit",
   309→    minute: "2-digit",
   310→  });
   311→}
   312→
   313→function initials(name: string | null): string {
   314→  return (name || "")
   315→    .split(/\s+/)
   316→    .filter(Boolean)
   317→    .slice(0, 2)
   318→    .map((s) => s[0]?.toUpperCase())
   319→    .join("");
   320→}

   399→// ─── Detail Dialog ───────────────────────────────────────────────
   400→function TicketDetailDialog({
   401→  ticket,
   402→  open,
   403→  onOpenChange,
   404→  onSave,
   405→  onAddMessage,
   406→  onAddNote,
   407→  onRunAi,
   408→  onDeleted,
   409→  saving,
   410→  categories,
   411→}: {
   412→  ticket: TicketDetail | null;
   413→  open: boolean;
   414→  onOpenChange: (v: boolean) => void;
   415→  onSave: (id: string, patch: TicketPatch) => Promise<boolean>;
   416→  onAddMessage: (id: string, content: string) => Promise<boolean>;
   417→  onAddNote: (id: string, content: string) => Promise<boolean>;
   418→  onRunAi: (id: string, action: "summary" | "suggest_reply" | "detect_priority" | "full_analysis") => Promise<AiResult | null>;
   419→  onDeleted: (id: string) => void;
   420→  saving: boolean;
   421→  categories: Category[];
   422→}) {
   423→  const [confirmDelete, setConfirmDelete] = React.useState(false);
   424→  const [activeTab, setActiveTab] = React.useState("messages");
   425→
   426→  // Subject editor (editable in header)
   427→  const [subjectDraft, setSubjectDraft] = React.useState("");
   428→  const [subjectDirty, setSubjectDirty] = React.useState(false);
   429→
   430→  // Sync subject draft when ticket changes
   431→  React.useEffect(() => {
   432→    if (ticket) {
   433→      setSubjectDraft(ticket.subject);
   434→      setSubjectDirty(false);
   435→    }
   436→  }, [ticket?.id, ticket?.subject]);
   437→
   438→  // Reset transient state when dialog closes
   439→  React.useEffect(() => {
   440→    if (!open) {
   441→      setConfirmDelete(false);
   442→      setActiveTab("messages");
   443→    }
   444→  }, [open]);
   445→
   446→  if (!ticket) return null;
   447→
   448→  const isOpen = ticket.status === "open";
   449→
   450→  async function commitSubject() {
   451→    if (!subjectDirty) return;
   452→    const trimmed = subjectDraft.trim();
   453→    if (!trimmed || trimmed === ticket!.subject) {
   454→      setSubjectDirty(false);
   455→      return;
   456→    }
   457→    const ok = await onSave(ticket!.id, { subject: trimmed });
   458→    if (!ok) {
   459→      setSubjectDraft(ticket!.subject);
   460→    }
   461→    setSubjectDirty(false);
   462→  }
   463→
   464→  return (
   465→    <Dialog open={open} onOpenChange={onOpenChange}>
   466→      <DialogContent className="sm:max-w-3xl max-h-[90vh] overflow-y-auto scroll-thin">
   467→        <DialogHeader>
   468→          <div className="flex items-start gap-3 pr-6">
   469→            <Avatar className="size-11 shrink-0">
   470→              <AvatarFallback className="brand-gradient text-white font-semibold">
   471→                {initials(ticket.customerName) || <TicketIcon size={18} />}
   472→              </AvatarFallback>
   473→            </Avatar>
   474→            <div className="min-w-0 flex-1">
   475→              <DialogTitle className="flex items-center gap-2 flex-wrap">
   476→                <Input
   477→                  value={subjectDraft}
   478→                  onChange={(e) => {
   479→                    setSubjectDraft(e.target.value);
   480→                    setSubjectDirty(true);
   481→                  }}
   482→                  onBlur={commitSubject}
   483→                  onKeyDown={(e) => {
   484→                    if (e.key === "Enter") {
   485→                      e.preventDefault();
   486→                      e.currentTarget.blur();
   487→                    }
   488→                    if (e.key === "Escape") {
   489→                      setSubjectDraft(ticket.subject);
   490→                      setSubjectDirty(false);
   491→                      (e.currentTarget as HTMLInputElement).blur();
   492→                    }
   493→                  }}
   494→                  className={cn(
   495→                    "h-7 px-1.5 text-base font-semibold border-transparent hover:border-border focus-visible:border-primary bg-transparent",
   496→                    !subjectDirty && "border-transparent"
   497→                  )}
   498→                />
   499→                {isOpen && (
   500→                  <span className="relative flex size-2 shrink-0">
   501→                    <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-primary opacity-75" />
   502→                    <span className="relative inline-flex size-2 rounded-full bg-primary" />
   503→                  </span>
   504→                )}
   505→              </DialogTitle>
   506→              <DialogDescription className="flex items-center gap-2 flex-wrap mt-1">
   507→                <PriorityBadge priority={ticket.priority} />
   508→                <StatusBadge status={ticket.status} />
   509→                <CategoryBadge category={ticket.category} />
   510→                {ticket.aiSentiment && (
   511→                  <SentimentBadge sentiment={ticket.aiSentiment} />
   512→                )}
   513→              </DialogDescription>
   514→              <div className="text-xs text-muted-foreground mt-1 flex items-center gap-1.5">
   515→                <UserCircle2 size={12} />
   516→                <span className="truncate">
   517→                  {ticket.customerName || "Anonymous"}
   518→                </span>
   519→                <span className="text-muted-foreground/50">·</span>
   520→                <Clock size={11} />
   521→                {relativeTime(ticket.createdAt)}
   522→              </div>
   523→            </div>
   524→          </div>
   525→        </DialogHeader>
   526→
   527→        <Tabs value={activeTab} onValueChange={setActiveTab} className="w-full">
   528→          <TabsList className="w-full">
   529→            <TabsTrigger value="messages" className="flex-1 gap-1.5">
   530→              <MessageCircle size={13} />
   531→              <span className="hidden sm:inline">Messages</span>
   532→              {ticket.messages.length > 0 && (
   533→                <Badge variant="secondary" className="ml-0.5 px-1.5 py-0 text-[10px] h-4 min-w-4 justify-center">
   534→                  {ticket.messages.length}
   535→                </Badge>
   536→              )}
   537→            </TabsTrigger>
   538→            <TabsTrigger value="notes" className="flex-1 gap-1.5">
   539→              <StickyNote size={13} />
   540→              <span className="hidden sm:inline">Notes</span>
   541→              {ticket.notes.length > 0 && (
   542→                <Badge variant="secondary" className="ml-0.5 px-1.5 py-0 text-[10px] h-4 min-w-4 justify-center">
   543→                  {ticket.notes.length}
   544→                </Badge>
   545→              )}
   546→            </TabsTrigger>
   547→            <TabsTrigger value="ai" className="flex-1 gap-1.5">
   548→              <Sparkles size={13} />
   549→              <span className="hidden sm:inline">AI Assistant</span>
   550→            </TabsTrigger>
   551→            <TabsTrigger value="details" className="flex-1 gap-1.5">
   552→              <CircleDot size={13} />
   553→              <span className="hidden sm:inline">Details</span>
   554→            </TabsTrigger>
   555→          </TabsList>
   556→
   557→          {/* ─── Messages tab ─── */}
   558→          <TabsContent value="messages" className="mt-4">
   559→            <MessagesTab
   560→              ticket={ticket}
   561→              onAddMessage={onAddMessage}
   562→              saving={saving}
   563→            />
   564→          </TabsContent>
   565→
   566→          {/* ─── Notes tab ─── */}
   567→          <TabsContent value="notes" className="mt-4">
   568→            <NotesTab
   569→              ticket={ticket}
   570→              onAddNote={onAddNote}
   571→              saving={saving}
   572→            />
   573→          </TabsContent>
   574→
   575→          {/* ─── AI Assistant tab ─── */}
   576→          <TabsContent value="ai" className="mt-4">
   577→            <AiAssistantTab
   578→              ticket={ticket}
   579→              onRunAi={onRunAi}
   580→              onAddMessage={onAddMessage}
   581→            />
   582→          </TabsContent>
   583→
   584→          {/* ─── Details tab ─── */}
   585→          <TabsContent value="details" className="mt-4 space-y-4">
   586→            <DetailsTab
   587→              ticket={ticket}
   588→              onSave={onSave}
   589→              saving={saving}
   590→              categories={categories}
   591→              onMarkResolved={() =>
   592→                onSave(ticket.id, { status: "resolved" })
   593→              }
   594→              onDelete={() => setConfirmDelete(true)}
   595→            />
   596→          </TabsContent>
   597→        </Tabs>
   598→
   599→        <DialogFooter className="gap-2 sm:gap-2">
   600→          <Button
   601→            variant="destructive"
   602→            size="sm"
   603→            onClick={() => setConfirmDelete(true)}
   604→            disabled={saving}
   605→            className="mr-auto"
   606→          >
   607→            <Trash2 size={14} />
   608→            Delete
   609→          </Button>
   610→          <Button
   611→            variant="outline"
   612→            size="sm"
   613→            onClick={() => onOpenChange(false)}
   614→            disabled={saving}
   615→          >
   616→            Close
   617→          </Button>
   618→        </DialogFooter>
   619→      </DialogContent>
   620→
   621→      <AlertDialog open={confirmDelete} onOpenChange={setConfirmDelete}>
   622→        <AlertDialogContent>
   623→          <AlertDialogHeader>

   904→// ─── AI Assistant tab ────────────────────────────────────────────
   905→function AiAssistantTab({
   906→  ticket,
   907→  onRunAi,
   908→  onAddMessage,
   909→}: {
   910→  ticket: TicketDetail;
   911→  onRunAi: (id: string, action: "summary" | "suggest_reply" | "detect_priority" | "full_analysis") => Promise<AiResult | null>;
   912→  onAddMessage: (id: string, content: string) => Promise<boolean>;
   913→}) {
   914→  // Local copies of stored AI values (updated when AI runs succeed)
   915→  const [summary, setSummary] = React.useState<string | null>(ticket.aiSummary);
   916→  const [suggestedReply, setSuggestedReply] = React.useState<string | null>(
   917→    ticket.aiSuggestedReply
   918→  );
   919→  const [priority, setPriority] = React.useState<TicketPriority | null>(
   920→    ticket.aiPriority
   921→  );
   922→  const [sentiment, setSentiment] = React.useState<TicketSentiment | null>(
   923→    ticket.aiSentiment
   924→  );
   925→  const [priorityReason, setPriorityReason] = React.useState<string | null>(
   926→    null
   927→  );
   928→  const [nextAction, setNextAction] = React.useState<string | null>(
   929→    ticket.aiNextAction
   930→  );
   931→
   932→  // Sync local state when ticket changes (e.g. opening a different ticket)
   933→  React.useEffect(() => {
   934→    setSummary(ticket.aiSummary);
   935→    setSuggestedReply(ticket.aiSuggestedReply);
   936→    setPriority(ticket.aiPriority);
   937→    setSentiment(ticket.aiSentiment);
   938→    setPriorityReason(null);
   939→    setNextAction(ticket.aiNextAction);
   940→  }, [
   941→    ticket.id,
   942→    ticket.aiSummary,
   943→    ticket.aiSuggestedReply,
   944→    ticket.aiPriority,
   945→    ticket.aiSentiment,
   946→    ticket.aiNextAction,
   947→  ]);
   948→
   949→  // Track which action is loading
   950→  const [loadingAction, setLoadingAction] = React.useState<string | null>(null);
   951→  const [sendingReply, setSendingReply] = React.useState(false);
   952→
   953→  async function run(
   954→    action: "summary" | "suggest_reply" | "detect_priority" | "full_analysis"
   955→  ) {
   956→    setLoadingAction(action);
   957→    try {
   958→      const result = await onRunAi(ticket.id, action);
   959→      if (!result) return;
   960→      if (result.error) {
   961→        toast.error(result.error);
   962→        return;
   963→      }
   964→      // Merge returned fields into local state
   965→      if (result.aiSummary !== undefined) setSummary(result.aiSummary ?? null);
   966→      if (result.aiSuggestedReply !== undefined)
   967→        setSuggestedReply(result.aiSuggestedReply ?? null);
   968→      if (result.aiPriority !== undefined)
   969→        setPriority(result.aiPriority ?? null);
   970→      if (result.aiSentiment !== undefined)
   971→        setSentiment(result.aiSentiment ?? null);
   972→      if (result.aiPriorityReason !== undefined)
   973→        setPriorityReason(result.aiPriorityReason ?? null);
   974→      if (result.aiNextAction !== undefined)
   975→        setNextAction(result.aiNextAction ?? null);
   976→      toast.success("AI analysis complete");
   977→    } finally {
   978→      setLoadingAction(null);
   979→    }
   980→  }
   981→
   982→  async function copyReply() {
   983→    if (!suggestedReply) return;
   984→    try {
   985→      await navigator.clipboard.writeText(suggestedReply);
   986→      toast.success("Copied reply to clipboard");
   987→    } catch {
   988→      toast.error("Couldn't access clipboard");
   989→    }
   990→  }
   991→
   992→  async function sendAsReply() {
   993→    if (!suggestedReply?.trim()) return;
   994→    setSendingReply(true);
   995→    try {
   996→      const ok = await onAddMessage(ticket.id, suggestedReply.trim());
   997→      if (ok) toast.success("Reply sent to customer");
   998→    } finally {
   999→      setSendingReply(false);
  1000→    }
  1001→  }
  1002→
  1003→  const actionBtns: {
  1004→    action: "summary" | "suggest_reply" | "detect_priority" | "full_analysis";
  1005→    label: string;
  1006→    icon: React.ComponentType<{ size?: number; className?: string }>;
  1007→    hint: string;
  1008→  }[] = [
  1009→    {
  1010→      action: "summary",
  1011→      label: "Generate Summary",
  1012→      icon: Sparkles,
  1013→      hint: "2-3 sentence overview of the issue",
  1014→    },
  1015→    {
  1016→      action: "suggest_reply",
  1017→      label: "Suggest Reply",
  1018→      icon: CornerUpLeft,
  1019→      hint: "Draft a professional reply",
  1020→    },
  1021→    {
  1022→      action: "detect_priority",
  1023→      label: "Detect Priority",
  1024→      icon: Gauge,
  1025→      hint: "Assess priority + sentiment",
  1026→    },
  1027→    {
  1028→      action: "full_analysis",
  1029→      label: "Full Analysis",
  1030→      icon: Wand2,
  1031→      hint: "Summary + reply + priority + next step",
  1032→    },
  1033→  ];
  1034→
  1035→  return (
  1036→    <div className="space-y-4">
  1037→      {/* Action buttons */}
  1038→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-2.5">
  1039→        {actionBtns.map((b) => {
  1040→          const isLoading = loadingAction === b.action;
  1041→          return (
  1042→            <Button
  1043→              key={b.action}
  1044→              variant="outline"
  1045→              className="h-auto justify-start gap-3 py-3 px-3.5"
  1046→              onClick={() => run(b.action)}
  1047→              disabled={!!loadingAction}
  1048→            >
  1049→              <div className="size-9 rounded-lg bg-primary/10 text-primary flex items-center justify-center shrink-0">
  1050→                {isLoading ? (
  1051→                  <Loader2 size={16} className="animate-spin" />
  1052→                ) : (
  1053→                  <b.icon size={16} />
  1054→                )}
  1055→              </div>
  1056→              <div className="text-left min-w-0">
  1057→                <div className="text-sm font-medium flex items-center gap-1.5">
  1058→                  {b.label}
  1059→                  {b.action === "full_analysis" && (
  1060→                    <Sparkles size={11} className="text-primary" />
  1061→                  )}
  1062→                </div>
  1063→                <div className="text-[11px] text-muted-foreground truncate">
  1064→                  {b.hint}
  1065→                </div>
  1066→              </div>
  1067→            </Button>
  1068→          );
  1069→        })}
  1070→      </div>
  1071→
  1072→      {/* Results */}
  1073→      <div className="space-y-3">

  1076→          title="AI Summary"

  1080→          value={summary}
  1081→        />
  1082→
  1083→        {/* Suggested Reply */}
  1084→        <div className="rounded-lg border border-primary/15 bg-primary/[0.03] p-3.5 space-y-2.5">
  1085→          <div className="flex items-center justify-between gap-2">
  1086→            <Label className="text-xs gap-1.5 text-foreground font-semibold">

  1094→                className="h-7 gap-1.5 text-xs"
  1095→                onClick={copyReply}

  1105→            <Skeleton className="h-20 w-full" />
