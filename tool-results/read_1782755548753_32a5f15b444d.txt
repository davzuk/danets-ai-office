   207→// ─── date helpers ────────────────────────────────────────────────
   208→function relativeTime(iso: string): string {
   209→  const d = new Date(iso);
   210→  const diff = Date.now() - d.getTime();
   211→  const sec = Math.floor(diff / 1000);
   212→  if (sec < 60) return "just now";
   213→  const min = Math.floor(sec / 60);
   214→  if (min < 60) return `${min}m ago`;
   215→  const hr = Math.floor(min / 60);
   216→  if (hr < 24) return `${hr}h ago`;
   217→  const day = Math.floor(hr / 24);
   218→  if (day < 7) return `${day}d ago`;
   219→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   220→}
   221→
   222→function fullDate(iso: string): string {
   223→  return new Date(iso).toLocaleString(undefined, {
   224→    month: "short",
   225→    day: "numeric",
   226→    year: "numeric",
   227→    hour: "2-digit",
   228→    minute: "2-digit",
   229→  });
   230→}

   232→function initials(name: string): string {
   233→  return name
   234→    .split(/\s+/)
   235→    .filter(Boolean)
   236→    .slice(0, 2)
   237→    .map((s) => s[0]?.toUpperCase())
   238→    .join("");
   239→}

   249→/** Contextual WhatsApp message body (NOT URL-encoded). */
   250→function buildWhatsAppMessage(lead: LeadContact): string {
   251→  const name = lead.name?.trim() || "there";
   252→  const service = lead.service?.trim();
   253→  const msg = lead.message?.trim();
   254→
   255→  const greeting = service
   256→    ? `Hi ${name}, thanks for your interest in ${service}. How can we help?`
   257→    : `Hi ${name}, thanks for reaching out. How can we help?`;
   258→
   259→  if (!msg) return greeting;
   260→  const snippet = msg.slice(0, 280);
   261→  const ellipsis = msg.length > 280 ? "…" : "";
   262→  return `${greeting}\n\nYou mentioned: "${snippet}${ellipsis}"`;
   263→}
   264→
   265→/** Contextual email subject line. */
   266→function buildEmailSubject(lead: LeadContact): string {
   267→  const service = lead.service?.trim();
   268→  return service
   269→    ? `Re: Your inquiry about ${service}`
   270→    : "Re: Your inquiry";
   271→}
   272→
   273→/** Contextual email body (plain text, line-break separated). */
   274→function buildEmailBody(lead: LeadContact): string {
   275→  const name = lead.name?.trim() || "there";
   276→  const service = lead.service?.trim();
   277→  const msg = lead.message?.trim();
   278→
   279→  const lines: string[] = [];
   280→  lines.push(`Hi ${name},`);
   281→  lines.push("");
   282→  lines.push(
   283→    service
   284→      ? `Thanks for reaching out about ${service}. We'd love to help!`
   285→      : `Thanks for reaching out. We'd love to help!`
   286→  );
   287→  if (msg) {
   288→    const snippet = msg.slice(0, 400);
   289→    const ellipsis = msg.length > 400 ? "…" : "";
   290→    lines.push("");
   291→    lines.push("You mentioned:");
   292→    lines.push(`"${snippet}${ellipsis}"`);
   293→  }
   294→  lines.push("");
   295→  lines.push(
   296→    "We'll get back to you shortly. Feel free to reply with any additional details."
   297→  );
   298→  lines.push("");
   299→  lines.push("Best regards,");
   300→  lines.push("The Danets Team");
   301→  return lines.join("\n");
   302→}

   304→/** Strip a phone string to digits-only for wa.me (intl format, no `+`). */
   305→function normalizePhoneForWhatsApp(phone: string): string {
   306→  return phone.replace(/\D/g, "");
   307→}
   308→
   309→/** Build a `https://wa.me/<digits>?text=<encoded>` URL, or null if no usable phone. */
   310→function buildWhatsAppUrl(lead: LeadContact): string | null {
   311→  if (!lead.phone) return null;
   312→  const digits = normalizePhoneForWhatsApp(lead.phone);
   313→  if (!digits) return null;
   314→  return `https://wa.me/${digits}?text=${encodeURIComponent(
   315→    buildWhatsAppMessage(lead)
   316→  )}`;
   317→}
   318→
   319→/** Build a `mailto:` URL with prefilled subject + body, or null if no email. */
   320→function buildMailtoUrl(lead: LeadContact): string | null {
   321→  if (!lead.email) return null;
   322→  const subject = encodeURIComponent(buildEmailSubject(lead));
   323→  const body = encodeURIComponent(buildEmailBody(lead));
   324→  return `mailto:${lead.email}?subject=${subject}&body=${body}`;
   325→}
   326→
   327→/** Build a `tel:` URL (whitespace stripped), or null if no phone. */
   328→function buildTelUrl(lead: LeadContact): string | null {
   329→  if (!lead.phone) return null;
   330→  return `tel:${lead.phone.replace(/\s+/g, "")}`;
   331→}
   332→
   333→/** Open WhatsApp in a new tab with a prefilled message + sonner feedback. */
   334→function openWhatsApp(lead: LeadContact) {
   335→  const url = buildWhatsAppUrl(lead);
   336→  if (!url) return;
   337→  window.open(url, "_blank", "noopener,noreferrer");
   338→  toast.success("Opening WhatsApp…");
   339→}
   340→
   341→/** Trigger the OS mailto handler with prefilled subject/body + sonner feedback. */
   342→function openEmail(lead: LeadContact) {
   343→  const url = buildMailtoUrl(lead);
   344→  if (!url) return;
   345→  window.location.href = url;
   346→  toast.success("Opening email client…");
   347→}
   348→
   349→/** Trigger the OS tel handler + sonner feedback. */
   350→function openCall(lead: LeadContact) {
   351→  const url = buildTelUrl(lead);
   352→  if (!url) return;
   353→  window.location.href = url;
   354→  toast.success("Initiating call…");
   355→}
