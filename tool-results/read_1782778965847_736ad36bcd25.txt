   243→function formatRelativeTime(iso: string | null): string | null {
   244→  if (!iso) return null;
   245→  const then = new Date(iso).getTime();
   246→  if (Number.isNaN(then)) return null;
   247→  const diff = Date.now() - then;
   248→  const sec = Math.round(diff / 1000);
   249→  if (sec < 60) return "just now";
   250→  const min = Math.round(sec / 60);
   251→  if (min < 60) return `${min}m ago`;
   252→  const hr = Math.round(min / 60);
   253→  if (hr < 24) return `${hr}h ago`;
   254→  const day = Math.round(hr / 24);
   255→  if (day < 30) return `${day}d ago`;
   256→  return new Date(iso).toLocaleDateString();
   257→}

   395→interface ConfigDialogProps {
   396→  category: Category;
   397→  integration?: Integration;
   398→  open: boolean;
   399→  onOpenChange: (v: boolean) => void;
   400→  onSaved: (integration: Integration) => void;

   403→
   404→function ConfigDialog({
   405→  category,
   406→  integration,
   407→  open,
   408→  onOpenChange,
   409→  onSaved,
   410→  onDeleted,
   411→}: ConfigDialogProps) {
   412→  const cfg = CATEGORY_CONFIG[category];
   413→  const options = PROVIDER_OPTIONS[category];
   414→
   415→  const [provider, setProvider] = useState<string>(integration?.provider ?? options[0].value);
   416→  const [apiKey, setApiKey] = useState("");
   417→  const [webhookUrl, setWebhookUrl] = useState(integration?.config?.webhookUrl ?? "");
   418→  const [phoneNumber, setPhoneNumber] = useState(integration?.config?.phoneNumber ?? "");
   419→  const [fromEmail, setFromEmail] = useState(integration?.config?.fromEmail ?? "");
   420→  const [smtpHost, setSmtpHost] = useState(integration?.config?.smtpHost ?? "");
   421→  const [senderId, setSenderId] = useState(integration?.config?.senderId ?? "");
   422→  const [model, setModel] = useState(integration?.config?.model ?? "");
   423→  const [notes, setNotes] = useState(integration?.notes ?? "");
   424→
   425→  const [saving, setSaving] = useState(false);
   426→  const [testing, setTesting] = useState(false);
   427→  const [deleteOpen, setDeleteOpen] = useState(false);
   428→  const [deleting, setDeleting] = useState(false);
   429→
   430→  // Reset state whenever the dialog opens (re-seed from integration).
   431→  useEffect(() => {
   432→    if (!open) return;
   433→    setProvider(integration?.provider ?? options[0].value);
   434→    setApiKey("");
   435→    setWebhookUrl(integration?.config?.webhookUrl ?? "");
   436→    setPhoneNumber(integration?.config?.phoneNumber ?? "");
   437→    setFromEmail(integration?.config?.fromEmail ?? "");
   438→    setSmtpHost(integration?.config?.smtpHost ?? "");
   439→    setSenderId(integration?.config?.senderId ?? "");
   440→    setModel(integration?.config?.model ?? "");
   441→    setNotes(integration?.notes ?? "");
   442→  }, [open, category, integration?.id]);
   443→
   444→  const providerOpt = options.find((o) => o.value === provider);
   445→  const needsApiKey = !!providerOpt?.needsApiKey;
   446→  const hasApiKey = !!integration?.config?.hasApiKey;
   447→
   448→  // Category-specific field visibility
   449→  const showPhone = category === "whatsapp";
   450→  const showFromEmail = category === "email";
   451→  const showSmtpHost = category === "email" && provider === "smtp";
   452→  const showSenderId = category === "sms";
   453→  const showWebhook = category === "crm" || (category === "whatsapp" && provider !== "manual");
   454→  const showModel = category === "ai_provider";
   455→
   456→  function buildConfig(): Record<string, unknown> | undefined {
   457→    const c: Record<string, unknown> = {};
   458→    if (apiKey.trim()) c.apiKey = apiKey.trim();
   459→    if (showWebhook) c.webhookUrl = webhookUrl.trim();
   460→    if (showPhone) c.phoneNumber = phoneNumber.trim();
   461→    if (showFromEmail) c.fromEmail = fromEmail.trim();
   462→    if (showSmtpHost) c.smtpHost = smtpHost.trim();
   463→    if (showSenderId) c.senderId = senderId.trim();
   464→    if (showModel && model.trim()) c.model = model.trim();
   465→    return c;
   466→  }
   467→
   468→  async function handleSave() {
   469→    if (!provider) {
   470→      toast.error("Please select a provider");
   471→      return;
   472→    }
   473→    setSaving(true);
   474→    try {
   475→      const config = buildConfig();
   476→      const payload: Record<string, unknown> = {
   477→        category,
   478→        provider,
   479→        name: providerLabel(category, provider),
   480→        status: integration?.status === "active" ? "active" : "inactive",
   481→        config,
   482→        notes: notes.trim() || null,
   483→      };
   484→      const res = await fetch("/api/integrations", {
   485→        method: "POST",
   486→        headers: { "Content-Type": "application/json" },
   487→        body: JSON.stringify(payload),
   488→      });
   489→      const data = await res.json();
   490→      if (!res.ok) throw new Error(data.error || "Failed to save integration");
   491→
   492→      // Refetch to get the masked config back
   493→      const listRes = await fetch("/api/integrations", { method: "GET" });
   494→      const listData = await listRes.json();
   495→      const fresh = (listData.integrations as Integration[])?.find(
   496→        (i) => i.category === category
   497→      );
   498→      if (fresh) onSaved(fresh);
   499→
   500→      toast.success("Integration saved", {
   501→        description: `${cfg.label} configured with ${providerLabel(category, provider)}.`,
   502→      });
   503→      onOpenChange(false);
   504→    } catch (e: unknown) {
   505→      const msg = e instanceof Error ? e.message : "Save failed";
   506→      toast.error("Save failed", { description: msg });
   507→    } finally {
   508→      setSaving(false);
   509→    }
   510→  }
   511→
   512→  async function handleTest() {
   513→    if (!integration) {
   514→      toast.error("Save the integration first to test the connection.");
   515→      return;
   516→    }
   517→    setTesting(true);
   518→    try {
   519→      const res = await fetch(`/api/integrations/${integration.id}`, {
   520→        method: "POST",
   521→      });
   522→      const data = await res.json();
   523→      if (!res.ok) throw new Error(data.error || "Test failed");
   524→
   525→      onSaved({
   526→        ...integration,
   527→        status: "active",
   528→        lastTestedAt: data.lastTestedAt ?? new Date().toISOString(),
   529→      });
   530→      toast.success("Connection test passed", {
   531→        description: data.message || `${cfg.label} is now marked as active.`,
   532→      });
   533→    } catch (e: unknown) {
   534→      const msg = e instanceof Error ? e.message : "Test failed";
   535→      toast.error("Connection test failed", { description: msg });
   536→      onSaved({ ...integration, status: "error" });
   537→    } finally {
   538→      setTesting(false);
   539→    }
   540→  }
   541→
   542→  async function handleDelete() {
   543→    if (!integration) return;
   544→    setDeleting(true);
   545→    try {
   546→      const res = await fetch(`/api/integrations/${integration.id}`, {
   547→        method: "DELETE",
   548→      });
   549→      const data = await res.json();
   550→      if (!res.ok) throw new Error(data.error || "Delete failed");
   551→      onDeleted(integration.id);
   552→      toast.success("Integration removed", {
   553→        description: `${cfg.label} configuration has been deleted.`,
   554→      });
   555→      setDeleteOpen(false);
   556→      onOpenChange(false);
   557→    } catch (e: unknown) {
   558→      const msg = e instanceof Error ? e.message : "Delete failed";
   559→      toast.error("Delete failed", { description: msg });
   560→    } finally {
   561→      setDeleting(false);
   562→    }
   563→  }
   564→
   565→  return (
   566→    <Dialog open={open} onOpenChange={onOpenChange}>
   567→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto scroll-thin">
   568→        <DialogHeader>
   569→          <DialogTitle className="flex items-center gap-2">
   570→            <div className={cn("size-7 rounded-md flex items-center justify-center", cfg.iconBg)}>
   571→              <cfg.icon className={cn("size-4", cfg.iconText)} />
   572→            </div>
   573→            Configure {cfg.label}
   574→          </DialogTitle>
   575→          <DialogDescription>{cfg.description}</DialogDescription>
   576→        </DialogHeader>
   577→
   578→        <div className="space-y-4 py-2">
   579→          {/* Provider */}
   580→          <div className="space-y-2">
   581→            <Label htmlFor="int-provider">Provider</Label>
   582→            <Select value={provider} onValueChange={setProvider}>
   583→              <SelectTrigger id="int-provider">
   584→                <SelectValue placeholder="Select a provider" />
   585→              </SelectTrigger>
   586→              <SelectContent>
   587→                {options.map((o) => (
   588→                  <SelectItem key={o.value} value={o.value}>
   589→                    {o.label}
   590→                  </SelectItem>
   591→                ))}
   592→              </SelectContent>
   593→            </Select>
   594→          </div>
   595→
   596→          {/* API Key */}
   597→          {needsApiKey ? (
   598→            <div className="space-y-2">
   599→              <Label htmlFor="int-apikey" className="flex items-center gap-1.5">
   600→                <KeyRound className="size-3.5" />
   601→                API Key
   602→              </Label>
   603→              <Input
   604→                id="int-apikey"
   605→                type="password"
   606→                autoComplete="off"
   607→                placeholder={
