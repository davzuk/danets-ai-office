   290→   290→      const res = await fetch(`/api/quotes/${quote.id}`);
   291→   291→      const data = await res.json();

   293→   293→      const latest = data.quote?.documents?.[0];

   295→   295→        setActiveDoc(latest);

   297→   297→        toast.error("No generated proposal found. Try Regenerate.");
   298→   298→        setViewOpen(false);

   300→   300→    } catch (e: any) {
   301→   301→      toast.error(e?.message || "Failed to load proposal");
   302→   302→      setViewOpen(false);
   303→   303→    } finally {
   304→   304→      setActiveLoading(false);

   326→   326→        clientName: quote.clientName,

   370→   370→  function downloadMarkdown(doc: GeneratedDocument) {
   371→   371→    const blob = new Blob([doc.content], { type: "text/markdown;charset=utf-8" });
   372→   372→    const url = URL.createObjectURL(blob);
   373→   373→    const a = document.createElement("a");
   374→   374→    const safeName =
   375→   375→      (doc.title || "proposal").replace(/[^a-z0-9-_]+/gi, "-").toLowerCase();
   376→   376→    a.href = url;
   377→   377→    a.download = `${safeName}.md`;
   378→   378→    document.body.appendChild(a);
   379→   379→    a.click();
   380→   380→    document.body.removeChild(a);
   381→   381→    URL.revokeObjectURL(url);
   382→   382→    toast.success("Downloaded markdown file");
   383→   383→  }
