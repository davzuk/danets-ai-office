   136→// Compute readable text color (black/white) for a given hex background
   137→function readableTextOn(hex: string): string {
   138→  if (!HEX_RE.test(hex)) return "#ffffff";
   139→  let h = hex.replace("#", "");
   140→  if (h.length === 3) {
   141→    h = h
   142→      .split("")
   143→      .map((c) => c + c)
   144→      .join("");
   145→  }
   146→  const r = parseInt(h.slice(0, 2), 16);
   147→  const g = parseInt(h.slice(2, 4), 16);
   148→  const b = parseInt(h.slice(4, 6), 16);
   149→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   150→  return luminance > 0.6 ? "#0b1220" : "#ffffff";
   151→}

   966→function Field({
   967→  label,
   968→  icon,
   969→  children,
   970→  required,
   971→  hint,
   972→  error,
   973→}: {
   974→  label: string;
   975→  icon?: React.ReactNode;
   976→  children: React.ReactNode;
   977→  required?: boolean;
   978→  hint?: string;
   979→  error?: string;
   980→}) {
   981→  return (
   982→    <div className="space-y-1.5">
   983→      <Label className="text-xs font-medium flex items-center gap-1.5">
   984→        {icon && <span className="text-muted-foreground">{icon}</span>}
   985→        {label}
   986→        {required && <span className="text-destructive">*</span>}
   987→      </Label>
   988→      {children}
   989→      {error ? (
   990→        <p className="text-xs text-destructive">{error}</p>
   991→      ) : hint ? (
   992→        <p className="text-xs text-muted-foreground">{hint}</p>
   993→      ) : null}
   994→    </div>
   995→  );
   996→}
