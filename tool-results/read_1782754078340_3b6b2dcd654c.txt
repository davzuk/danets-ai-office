   275→   275→function formatDate(iso: string) {
   276→   276→  try {
   277→   277→    const d = new Date(iso);
   278→   278→    return d.toLocaleDateString(undefined, {
   279→   279→      year: "numeric",
   280→   280→      month: "short",
   281→   281→      day: "numeric",
   282→   282→    });
   283→   283→  } catch {
   284→   284→    return iso;
   285→   285→  }
   286→   286→}

   288→   288→function relativeTime(iso: string) {
   289→   289→  try {
   290→   290→    const d = new Date(iso).getTime();
   291→   291→    const now = Date.now();
   292→   292→    const diff = Math.max(0, now - d);
   293→   293→    const mins = Math.floor(diff / 60_000);
   294→   294→    if (mins < 1) return "just now";
   295→   295→    if (mins < 60) return `${mins}m ago`;
   296→   296→    const hrs = Math.floor(mins / 60);
   297→   297→    if (hrs < 24) return `${hrs}h ago`;
   298→   298→    const days = Math.floor(hrs / 24);
   299→   299→    if (days < 30) return `${days}d ago`;
   300→   300→    return formatDate(iso);
   301→   301→  } catch {
   302→   302→    return formatDate(iso);
   303→   303→  }
   304→   304→}

   448→   448→function hashCategoryToKey(category: string): keyof typeof CATEGORY_PALETTE {
   449→   449→  let hash = 0;
   450→   450→  const str = category.toLowerCase();
   451→   451→  for (let i = 0; i < str.length; i++) {
   452→   452→    hash = (hash << 5) - hash + str.charCodeAt(i);
   453→   453→    hash |= 0;
   454→   454→  }
   455→   455→  return CATEGORY_HASH_KEYS[Math.abs(hash) % CATEGORY_HASH_KEYS.length];
   456→   456→}

   463→   463→}
