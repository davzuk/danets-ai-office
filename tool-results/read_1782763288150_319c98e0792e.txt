   293→function formatDate(iso: string) {
   294→  try {
   295→    const d = new Date(iso);
   296→    return d.toLocaleDateString(undefined, {
   297→      year: "numeric",
   298→      month: "short",
   299→      day: "numeric",
   300→    });
   301→  } catch {
   302→    return iso;
   303→  }
   304→}

   306→function relativeTime(iso: string) {
   307→  try {
   308→    const d = new Date(iso).getTime();
   309→    const now = Date.now();
   310→    const diff = Math.max(0, now - d);
   311→    const mins = Math.floor(diff / 60_000);
   312→    if (mins < 1) return "just now";
   313→    if (mins < 60) return `${mins}m ago`;
   314→    const hrs = Math.floor(mins / 60);
   315→    if (hrs < 24) return `${hrs}h ago`;
   316→    const days = Math.floor(hrs / 24);
   317→    if (days < 30) return `${days}d ago`;
   318→    return formatDate(iso);
   319→  } catch {
   320→    return formatDate(iso);
   321→  }
   322→}

   474→function hashCategoryToKey(category: string): keyof typeof CATEGORY_PALETTE {
   475→  let hash = 0;
   476→  const str = category.toLowerCase();
   477→  for (let i = 0; i < str.length; i++) {
   478→    hash = (hash << 5) - hash + str.charCodeAt(i);
   479→    hash |= 0;
   480→  }
   481→  return CATEGORY_HASH_KEYS[Math.abs(hash) % CATEGORY_HASH_KEYS.length];
   482→}
   483→
   484→function getCategoryColorKey(
   485→  category: string
   486→): keyof typeof CATEGORY_PALETTE {
   487→  const key = category.trim().toLowerCase();
   488→  return CATEGORY_NAME_MAP[key] ?? hashCategoryToKey(category);
   489→}

   491→function getCategoryColor(category: string): CategoryColorClasses {
   492→  return CATEGORY_PALETTE[getCategoryColorKey(category)];
   493→}

   783→  /** Run a KB test search against the TF-IDF retrieval system. Aborts any
   784→   * in-flight request so the latest query wins. Errors are toasted (per spec)
   785→   * and also surfaced inline in the dialog. */
   786→  const runTestSearch = useCallback(async (q: string) => {
   787→    const query = q.trim();
   788→    if (!query) return;
   789→    testSearchAbort.current?.abort();
   790→    const ac = new AbortController();
   791→    testSearchAbort.current = ac;
   792→    setTestSearching(true);
   793→    setTestError(null);
   794→    try {
   795→      const res = await fetch("/api/knowledge-base/test-search", {
   796→        method: "POST",
   797→        headers: { "Content-Type": "application/json" },
   798→        body: JSON.stringify({ query }),
   799→        signal: ac.signal,
   800→      });
   801→      const data = await res.json().catch(() => null);
   802→      if (!res.ok) {
   803→        throw new Error(data?.error || "Search failed");
   804→      }
   805→      setTestResults(data as TestSearchResults);
   806→    } catch (e: any) {
   807→      if (e?.name === "AbortError") return;
   808→      const msg = e?.message || "Could not run search";
   809→      setTestError(msg);
   810→      toast.error(msg);
   811→    } finally {
   812→      if (!ac.signal.aborted) setTestSearching(false);
   813→    }
   814→  }, []);

   817→  // when the input is emptied, and is bypassed by manual submit / chip clicks.
   818→  useEffect(() => {

   824→    }
   825→    if (!query) {
   826→      setTestResults(null);
   827→      setTestError(null);
   828→      setTestSearching(false);
   829→      return;
   830→    }
   831→    setTestSearching(true);
   832→    testSearchTimer.current = setTimeout(() => {
   833→      testSearchTimer.current = null;
   834→      runTestSearch(query);
   835→    }, 400);
   836→    return () => {
   837→      if (testSearchTimer.current) {
   838→        clearTimeout(testSearchTimer.current);
   839→        testSearchTimer.current = null;
   840→      }
   841→    };
   842→  }, [testQuery, testSearchOpen, runTestSearch]);

  1732→function TestSearchResultCard({ match }: { match: TestSearchMatch }) {

  1734→  // Ranks beyond 5 (backend caps at 5) floor at 20%.
  1735→  const relevance = Math.max(20, 100 - (match.rank - 1) * 20);
  1736→  const barColor =
  1737→    relevance >= 80
  1738→      ? "bg-emerald-500"
  1739→      : relevance >= 60
  1740→      ? "bg-primary"
  1741→      : relevance >= 40
  1742→      ? "bg-amber-500"
  1743→      : "bg-rose-500";
  1744→  const color = getCategoryColor(match.category);
