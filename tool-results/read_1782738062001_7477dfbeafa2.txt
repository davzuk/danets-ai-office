   136→   136→// Compute readable text color (black/white) for a given hex background
   137→   137→function readableTextOn(hex: string): string {
   138→   138→  if (!HEX_RE.test(hex)) return "#ffffff";
   139→   139→  let h = hex.replace("#", "");
   140→   140→  if (h.length === 3) {
   141→   141→    h = h
   142→   142→      .split("")
   143→   143→      .map((c) => c + c)
   144→   144→      .join("");
   145→   145→  }
   146→   146→  const r = parseInt(h.slice(0, 2), 16);
   147→   147→  const g = parseInt(h.slice(2, 4), 16);
   148→   148→  const b = parseInt(h.slice(4, 6), 16);
   149→   149→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   150→   150→  return luminance > 0.6 ? "#0b1220" : "#ffffff";
   151→   151→}
